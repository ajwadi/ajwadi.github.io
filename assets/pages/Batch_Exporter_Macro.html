CATIA Batch Exporter

Exports every part of an assembly, the active document, or a whole folder of CATIA files to STEP, IGES, STL, CGR or 3DXML, with a naming rule built from the part number, revision and date. Two code blocks to paste: the form and the module.

What it does

  • Three sources. Every part in the active product, the active document only, or every CATPart (and optionally CATProduct) in a folder you choose.
  • Five formats. STEP, IGES, STL, CGR and 3DXML. STL is for parts only.
  • Naming rule. Files are named from the part number, with the revision and today's date added if you tick those boxes, for example BRKT-001_B_20261011.stp.
  • Each part once. A part used ten times in the assembly is exported one time.
  • Progress on the status line, with a count of exported and failed files at the end. Existing files with the same name are replaced.
  • English / Korean button in the top right corner. The language is remembered for next time.
  • Credit image centered at the bottom, stored inside the module as text, the same as your other macros.
  • Works beside your other macros. Every shared name starts with BEx, so it does not clash with the other macros in the same VBA project.

Setup

  1. In the VBA editor, use Insert, UserForm. In the Properties window set its (Name) to frmBatchExporter. Leave the form empty in the designer.
  2. Open the form's code (right-click the form, View Code), select all, delete, and paste the Form code below.
  3. Use Insert, Module. In the Properties window set its (Name) to BatchExporter, then paste the Module code below.
  4. Use Debug, Compile VBAProject.
  5. Open a CATProduct or CATPart (not needed for the folder option), then run CATMain from the BatchExporter module.

First run: this code was written and checked for structure, but not run inside CATIA. Try it on a small assembly with STEP first, into an empty folder.

Module code

BatchExporter

The long block of letters and numbers at the bottom is the credit image. Do not edit it.

Option Explicit

'=========================================================
' CATIA BATCH EXPORTER - MAIN MODULE
' Module name: BatchExporter
' Run CATMain to open the window.
'=========================================================

Private Const REG_APP As String = "CatiaBatchExporter"
Private Const REG_SEC As String = "Settings"

' "EN" or "KO"
Public gBExLang As String


'=========================================================
' ENTRY POINT
'=========================================================

Sub CATMain()

    BExLoadLanguage
    frmBatchExporter.Show

End Sub

'=========================================================
' EXPORT
'=========================================================

Public Function BExRun( _
        ByVal frm As Object, _
        ByVal actionId As Long, _
        ByRef message As String) As Boolean

    Dim docs As Object
    Dim keys As Variant
    Dim d As Object
    Dim homeWindow As Object
    Dim i As Long
    Dim ext As String
    Dim outFolder As String
    Dim source As Long
    Dim done As Long
    Dim failed As Long
    Dim fileName As String
    Dim wantAsm As Boolean
    Dim kind As String

    BExRun = False

    outFolder = WithSlash(CtlText(frm, "txtOut"))

    If Not FolderExists(outFolder) Then
        message = BExTxt("msgNoOut")
        Exit Function
    End If

    Select Case CtlIndex(frm, "cboFormat")
        Case 1: ext = "igs"
        Case 2: ext = "stl"
        Case 3: ext = "cgr"
        Case 4: ext = "3dxml"
        Case Else: ext = "stp"
    End Select

    source = CtlIndex(frm, "cboSource")
    wantAsm = CtlOn(frm, "chkAsm") And ext <> "stl"

    If source = 2 Then
        BExRun = ExportFolder(frm, outFolder, ext, wantAsm, message)
        Exit Function
    End If

    kind = BExDocKind()

    If kind <> "Product" And kind <> "Part" Then
        message = BExTxt("msgNeedDoc")
        Exit Function
    End If

    Set d = CATIA.ActiveDocument
    Set docs = CreateObject("Scripting.Dictionary")

    If source = 0 And kind = "Product" Then

        ' Parts must be loaded in design mode to be exported
        On Error Resume Next
        d.Product.ApplyWorkMode 2
        On Error GoTo 0

        CollectDocs d.Product, docs, True, wantAsm

        If wantAsm Then AddDoc docs, d

    ElseIf kind = "Product" And ext = "stl" Then

        message = BExTxt("msgStlParts")
        Exit Function

    Else

        AddDoc docs, d

    End If

    If docs.Count = 0 Then
        message = BExTxt("msgNothing")
        Exit Function
    End If

    keys = docs.Keys

    Set homeWindow = CATIA.ActiveWindow

    CATIA.DisplayFileAlerts = False

    For i = 0 To docs.Count - 1

        Set d = docs.Item(keys(i))

        fileName = BuildName(d, frm) & "." & ext

        Say frm, BExFmt(BExTxt("stItem"), CStr(i + 1), CStr(docs.Count)) & " " & fileName

        If ExportOne(d, outFolder & fileName, ext) Then
            done = done + 1
        Else
            failed = failed + 1
        End If

    Next i

    CATIA.DisplayFileAlerts = True

    On Error Resume Next
    homeWindow.Activate
    On Error GoTo 0

    message = BExFmt(BExTxt("msgDone"), CStr(done), CStr(failed))

    BExRun = True

End Function


' File name without extension: part number, then the
' optional revision and date
Private Function BuildName(ByVal d As Object, ByVal frm As Object) As String

    Dim baseName As String
    Dim rev As String

    On Error Resume Next

    baseName = d.Product.PartNumber
    rev = d.Product.Revision

    On Error GoTo 0

    If Len(baseName) = 0 Then baseName = StripExt(d.Name)

    If CtlOn(frm, "chkRev") And Len(rev) > 0 Then baseName = baseName & "_" & rev

    If CtlOn(frm, "chkDate") Then baseName = baseName & "_" & Format$(Date, "yyyymmdd")

    BuildName = CleanName(baseName)

End Function


' Exports one loaded document. Returns True when the file exists afterwards.
Private Function ExportOne( _
        ByVal d As Object, _
        ByVal filePath As String, _
        ByVal ext As String) As Boolean

    Dim ownWindow As Boolean
    Dim before As Long

    On Error GoTo Failed

    DeleteFile filePath

    ' A part that is only loaded inside an assembly has no window of
    ' its own. STL export needs one, so it is opened and closed again.
    If ext = "stl" And Len(d.Path) > 0 Then
        before = CATIA.Windows.Count
        CATIA.Documents.Open d.FullName
        ownWindow = (CATIA.Windows.Count > before)
    End If

    d.ExportData filePath, ext

    If ownWindow Then CATIA.ActiveWindow.Close

    ExportOne = FileExists(filePath)

    Exit Function

Failed:

    On Error Resume Next

    If ownWindow Then CATIA.ActiveWindow.Close

End Function


' Opens every CATIA file of a folder, exports it and closes it again
Private Function ExportFolder( _
        ByVal frm As Object, _
        ByVal outFolder As String, _
        ByVal ext As String, _
        ByVal wantAsm As Boolean, _
        ByRef message As String) As Boolean

    Dim fso As Object
    Dim f As Object
    Dim paths As New Collection
    Dim srcFolder As String
    Dim e As String
    Dim i As Long
    Dim done As Long
    Dim failed As Long

    ExportFolder = False

    srcFolder = WithSlash(CtlText(frm, "txtSource"))

    If Not FolderExists(srcFolder) Then
        message = BExTxt("msgNoSrc")
        Exit Function
    End If

    Set fso = CreateObject("Scripting.FileSystemObject")

    For Each f In fso.GetFolder(srcFolder).Files

        e = LCase$(fso.GetExtensionName(f.Name))

        If e = "catpart" Then paths.Add CStr(f.Path)
        If e = "catproduct" And wantAsm Then paths.Add CStr(f.Path)

    Next f

    If paths.Count = 0 Then
        message = BExTxt("msgNothing")
        Exit Function
    End If

    CATIA.DisplayFileAlerts = False

    For i = 1 To paths.Count

        Say frm, BExFmt(BExTxt("stItem"), CStr(i), CStr(paths.Count)) & " " & FileNameOf(CStr(paths.Item(i)))

        If ExportFile(CStr(paths.Item(i)), outFolder, ext, frm) Then
            done = done + 1
        Else
            failed = failed + 1
        End If

    Next i

    CATIA.DisplayFileAlerts = True

    message = BExFmt(BExTxt("msgDone"), CStr(done), CStr(failed))

    ExportFolder = True

End Function


Private Function ExportFile( _
        ByVal filePath As String, _
        ByVal outFolder As String, _
        ByVal ext As String, _
        ByVal frm As Object) As Boolean

    Dim d As Object
    Dim wasOpen As Boolean
    Dim target As String

    On Error GoTo Failed

    wasOpen = IsDocOpen(FileNameOf(filePath))

    Set d = CATIA.Documents.Open(filePath)

    target = outFolder & BuildName(d, frm) & "." & ext

    DeleteFile target

    d.ExportData target, ext

    ExportFile = FileExists(target)

    If Not wasOpen Then d.Close

    Exit Function

Failed:

    On Error Resume Next

    If Not wasOpen Then
        If Not d Is Nothing Then d.Close
    End If

End Function

' "" = nothing open, otherwise Product, Part, Drawing or Other
Public Function BExDocKind() As String

    Dim t As String

    On Error Resume Next

    If CATIA.Documents.Count > 0 Then t = TypeName(CATIA.ActiveDocument)

    On Error GoTo 0

    Select Case t
        Case "ProductDocument": BExDocKind = "Product"
        Case "PartDocument": BExDocKind = "Part"
        Case "DrawingDocument": BExDocKind = "Drawing"
        Case "": BExDocKind = ""
        Case Else: BExDocKind = "Other"
    End Select

End Function


Public Function BExDocName() As String

    On Error Resume Next

    BExDocName = CATIA.ActiveDocument.Name

End Function

' Replaces characters that are not allowed in a file name
Private Function CleanName(ByVal rawName As String) As String

    Dim i As Long
    Dim ch As String
    Dim code As Long
    Dim result As String

    For i = 1 To Len(rawName)

        ch = Mid$(rawName, i, 1)
        code = AscW(ch)

        If InStr("\/:*?""<>|", ch) > 0 Or (code >= 0 And code < 32) Then
            result = result & "_"
        Else
            result = result & ch
        End If

    Next i

    If Len(result) = 0 Then result = "Unnamed"

    CleanName = result

End Function


Private Function StripExt(ByVal fileName As String) As String

    Dim p As Long

    p = InStrRev(fileName, ".")

    If p > 1 Then
        StripExt = Left$(fileName, p - 1)
    Else
        StripExt = fileName
    End If

End Function

' A number written with a decimal point on every PC
Private Function Num(ByVal numberValue As Variant, Optional ByVal decimals As Long = 3) As String

    Dim pattern As String

    On Error Resume Next

    pattern = "0"

    If decimals > 0 Then pattern = "0." & String$(decimals, "0")

    Num = Replace(Format$(CDbl(numberValue), pattern), ",", ".")

End Function


' Saves text as UTF-8 so Excel shows Korean correctly
Private Function WriteTextFile(ByVal filePath As String, ByVal content As String) As Boolean

    Dim stm As Object

    On Error GoTo Failed

    Set stm = CreateObject("ADODB.Stream")

    stm.Type = 2
    stm.Charset = "utf-8"
    stm.Open
    stm.WriteText content
    stm.SaveToFile filePath, 2
    stm.Close

    WriteTextFile = True

    Exit Function

Failed:

    stm.Close

End Function

    stm.Charset = "utf-8"
    stm.Open

    ReadTextFile = stm.ReadText

    stm.Close

    On Error Resume Next
    stm.Close

End Function


' Splits one CSV line into its fields
Private Function CsvSplit(ByVal lineText As String) As Collection

    Dim fields As New Collection
    Dim i As Long
    Dim ch As String
    Dim cur As String
    Dim inQuotes As Boolean

    i = 1

    Do While i <= Len(lineText)

        ch = Mid$(lineText, i, 1)

        If inQuotes Then
            If ch = """" Then
                If Mid$(lineText, i + 1, 1) = """" Then
                    cur = cur & """"
                    i = i + 1
                Else
                    inQuotes = False
                End If
            Else
                cur = cur & ch
            End If
        ElseIf ch = """" Then
            inQuotes = True
        ElseIf ch = "," Then
            fields.Add cur
            cur = ""
        Else
            cur = cur & ch
        End If

        i = i + 1

    Loop

    fields.Add cur

    Set CsvSplit = fields

End Function

    If Right$(TypeName(d), 8) = "Document" Then Set DocOf = d

End Function


Private Sub AddDoc(ByVal docs As Object, ByVal d As Object)

    Dim key As String

    key = UCase$(d.FullName)

    If Not docs.Exists(key) Then docs.Add key, d

End Sub


' Fills docs (a Dictionary) with every loaded document under prod
Private Sub CollectDocs( _
        ByVal prod As Object, _
        ByVal docs As Object, _
        ByVal wantParts As Boolean, _
        ByVal wantProducts As Boolean)

    Dim kids As Object
    Dim child As Object
    Dim d As Object
    Dim kind As String
    Dim i As Long

    On Error Resume Next
    Set kids = prod.Products
    On Error GoTo 0

    If kids Is Nothing Then Exit Sub

    For i = 1 To kids.Count

        Set child = kids.Item(i)
        Set d = DocOf(child)

        If Not d Is Nothing Then

            kind = TypeName(d)

            If kind = "PartDocument" And wantParts Then AddDoc docs, d
            If kind = "ProductDocument" And wantProducts Then AddDoc docs, d

            CollectDocs child, docs, wantParts, wantProducts

        End If

    Next i

End Sub

' Turns \uXXXX codes into real characters
Private Function U(ByVal coded As String) As String

    Dim i As Long
    Dim result As String

    i = 1

    Do While i <= Len(coded)

        If Mid$(coded, i, 2) = "\u" And i + 5 <= Len(coded) Then
            result = result & ChrW$(CLng("&H" & Mid$(coded, i + 2, 4)))
            i = i + 6
        Else
            result = result & Mid$(coded, i, 1)
            i = i + 1
        End If

    Loop

    U = result

End Function

' LOGO
' The logo picture is stored below as text.
' The form shows it with:
'     BExLoadLogo Me.Controls("imgLogo")
'=========================================================

' Puts the logo into an Image control.
' Returns False if the picture could not be loaded.
Public Function BExLoadLogo(ByVal target As Object) As Boolean

    Dim data() As Byte
    Dim filePath As String
    Dim fileNo As Integer

    BExLoadLogo = False

    On Error GoTo Failed

    data = HexToBytes(LogoHex())

    ' VBA can only load a picture from a file, so the data is
    ' written to the Windows temp folder, loaded, then deleted.
    filePath = Environ$("TEMP") & "\catia_macro_logo.gif"

    If Len(Dir$(filePath)) > 0 Then Kill filePath

    fileNo = FreeFile

    Open filePath For Binary Access Write As #fileNo
    Put #fileNo, , data
    Close #fileNo

    Set target.Picture = LoadPicture(filePath)

    BExLoadLogo = True

    On Error Resume Next
    Kill filePath

    Exit Function

Failed:

    On Error Resume Next
    Close #fileNo

End Function


' Turns pairs of hex characters back into bytes
Private Function HexToBytes(ByVal hexText As String) As Byte()

    Dim count As Long
    Dim i As Long
    Dim data() As Byte

    count = Len(hexText) \ 2

    ReDim data(0 To count - 1)

    For i = 0 To count - 1
        data(i) = CByte("&H" & Mid$(hexText, i * 2 + 1, 2))
    Next i

    HexToBytes = data

End Function

Known limitations

  • For STL, each part is opened in its own window for a moment and closed again, so the screen flickers. Parts that were never saved are skipped for STL.
  • The assembly is switched to design mode so every part is loaded. On a large assembly this can take a while.
  • Export quality settings (STEP AP203 or AP214, STL sag and so on) come from Tools, Options. The macro does not change them.
  • 3D PDF is not included because it needs a separate CATIA license.
  • In folder mode, files in sub-folders are not included.
  • The window is modal, so CATIA cannot be clicked while it is open. Anything that must be selected in CATIA has to be selected before running the macro.
  • If the credit image cannot be loaded, for example when the Windows temp folder is not writable, the window opens without it.
Prepared for CATIA V5 R21 and later, VBA. Exports parts and assemblies to neutral formats in one run.
