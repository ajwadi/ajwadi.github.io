CATIA V5 R21 BOM + Physical Properties Exporter

CSV export macro for BOM, material, mass, volume, density, thickness, and assembly metadata

Project Structure

VBAProject │ └── Modules └── BOMPhysicalProperties └── CATMain

Recommended setup: create one standard VBA module named BOMPhysicalProperties and paste the full code from this page into it.

What This Macro Exports

ColumnDescription
Part NumberCATIA product part number.
QuantityNumber of matching occurrences in the assembly.
TypePart or Assembly.
MaterialApplied CATIA material when available.
Mass Each (kg)Mass of one component from CATIA inertia properties.
Total Mass (kg)Mass Each × Quantity.
Volume (mm³)Component volume converted to cubic millimetres.
Density (kg/m³)Density returned by CATIA inertia properties.
Thickness (mm)Defined thickness parameter if found, otherwise a thin-wall estimate.
Thickness MethodShows whether thickness came from a parameter or geometric estimate.
Thickness StatusExplains confidence or whether thickness is unavailable.
Nomenclature / Revision / DescriptionStandard CATIA metadata.
Example Instance / Parent AssemblyAssembly-tree context.
File PathReferenced CATPart or CATProduct path when accessible.

Important Thickness Behavior

Preferred method: if the CATPart contains an explicit thickness parameter, the macro reports it and labels the method Parameter.

Fallback: if no thickness parameter is found, the macro estimates thickness using 2 × Volume / Wet Area. This is most meaningful for thin, fairly constant-wall parts and should be verified for blocks, castings, ribs, or variable-wall geometry.

Full VBA Code

Module: BOMPhysicalProperties 
Option Explicit

'=========================================================
' CATIA BOM + PHYSICAL PROPERTIES CSV EXPORTER
' CATIA V5 R21
'=========================================================

Public Sub CATMain()

    Dim productDoc As ProductDocument
    Dim rootProduct As Product
    Dim bomData As Object
    Dim savePath As String
    Dim fileNumber As Integer
    Dim key As Variant
    Dim itemData As Variant
    Dim itemNumber As Long

    If CATIA.Documents.Count = 0 Then
        MsgBox "Please open a CATProduct before running the report.", _
               vbExclamation, "CATIA BOM + Properties"
        Exit Sub
    End If

    If TypeName(CATIA.ActiveDocument) <> "ProductDocument" Then
        MsgBox "The active document must be a CATProduct.", _
               vbExclamation, "CATIA BOM + Properties"
        Exit Sub
    End If

    Set productDoc = CATIA.ActiveDocument
    Set rootProduct = productDoc.Product

    Set bomData = CreateObject("Scripting.Dictionary")

    Call ReadProductTree(rootProduct, bomData)

    If bomData.Count = 0 Then
        MsgBox "No components were found.", _
               vbInformation, "CATIA BOM + Properties"
        Exit Sub
    End If

    savePath = CATIA.FileSelectionBox( _
        "Save CATIA BOM + Physical Properties", _
        "*.csv", _
        1)

    If savePath = "" Then Exit Sub

    If LCase(Right(savePath, 4)) <> ".csv" Then
        savePath = savePath & ".csv"
    End If

    On Error GoTo WriteError

    fileNumber = FreeFile
    Open savePath For Output As #fileNumber

    Print #fileNumber, CSVLine(Array( _
        "CATIA BOM + Physical Properties Report", _
        "", "", "", "", "", "", "", "", "", "", "", "", "", "", "", ""))

    Print #fileNumber, CSVLine(Array( _
        "Assembly", rootProduct.PartNumber, _
        "", "", "", "", "", "", "", "", "", "", "", "", "", "", ""))

    Print #fileNumber, CSVLine(Array( _
        "CATProduct", productDoc.Name, _
        "", "", "", "", "", "", "", "", "", "", "", "", "", "", ""))

    Print #fileNumber, CSVLine(Array( _
        "Generated", CStr(Now), _
        "", "", "", "", "", "", "", "", "", "", "", "", "", "", ""))

    Print #fileNumber, ""

    Print #fileNumber, CSVLine(Array( _
        "Item", _
        "Part Number", _
        "Quantity", _
        "Type", _
        "Material", _
        "Mass Each (kg)", _
        "Total Mass (kg)", _
        "Volume (mm3)", _
        "Density (kg/m3)", _
        "Thickness (mm)", _
        "Thickness Method", _
        "Thickness Status", _
        "Nomenclature", _
        "Revision", _
        "Description", _
        "Example Instance", _
        "Parent Assembly", _
        "File Path"))

    itemNumber = 1

    For Each key In bomData.Keys
        itemData = bomData(key)

        Print #fileNumber, CSVLine(Array( _
            CStr(itemNumber), _
            CStr(itemData(0)), _
            CStr(itemData(1)), _
            CStr(itemData(2)), _
            CStr(itemData(3)), _
            FormatCSVNumber(itemData(4)), _
            FormatCSVNumber(itemData(5)), _
            FormatCSVNumber(itemData(6)), _
            FormatCSVNumber(itemData(7)), _
            FormatCSVNumber(itemData(8)), _
            CStr(itemData(9)), _
            CStr(itemData(10)), _
            CStr(itemData(11)), _
            CStr(itemData(12)), _
            CStr(itemData(13)), _
            CStr(itemData(14)), _
            CStr(itemData(15)), _
            CStr(itemData(16))))

        itemNumber = itemNumber + 1
    Next key

    Close #fileNumber

    MsgBox "Report exported successfully." & _
           vbCrLf & vbCrLf & savePath, _
           vbInformation, "CATIA BOM + Properties"

    Exit Sub

WriteError:

    On Error Resume Next
    Close #fileNumber

    MsgBox "The CSV could not be written." & _
           vbCrLf & vbCrLf & _
           "Error " & CStr(Err.Number) & _
           vbCrLf & Err.Description, _
           vbCritical, "CATIA BOM + Properties"

End Sub


Private Sub ReadProductTree( _
    ByVal parentProduct As Product, _
    ByRef bomData As Object)

    Dim children As Products
    Dim child As Product
    Dim i As Long

    Dim partNumber As String
    Dim instanceName As String
    Dim parentName As String
    Dim nomenclature As String
    Dim revision As String
    Dim description As String
    Dim componentType As String
    Dim filePath As String
    Dim bomKey As String
    Dim materialName As String

    Dim massEach As Double
    Dim totalMass As Double
    Dim volumeMM3 As Double
    Dim density As Double
    Dim thicknessMM As Double
    Dim thicknessMethod As String
    Dim thicknessStatus As String
    Dim quantity As Long
    Dim itemData As Variant

    Set children = parentProduct.Products
    parentName = SafePartNumber(parentProduct)

    For i = 1 To children.Count

        Set child = children.Item(i)

        partNumber = SafePartNumber(child)
        instanceName = SafeInstanceName(child)
        nomenclature = SafeNomenclature(child)
        revision = SafeRevision(child)
        description = SafeDescription(child)
        filePath = GetProductFilePath(child)

        If child.Products.Count > 0 Then
            componentType = "Assembly"
        Else
            componentType = "Part"
        End If

        materialName = ""
        massEach = 0#
        volumeMM3 = 0#
        density = 0#
        thicknessMM = 0#
        thicknessMethod = ""
        thicknessStatus = ""

        If componentType = "Part" Then

            materialName = GetMaterialName(child)

            Call GetPhysicalProperties( _
                child, massEach, volumeMM3, density)

            Call GetThickness( _
                child, volumeMM3, thicknessMM, _
                thicknessMethod, thicknessStatus)

        Else

            thicknessStatus = "N/A - Assembly"

        End If

        If filePath <> "" Then
            bomKey = UCase(filePath) & "|" & UCase(componentType)
        Else
            bomKey = UCase(partNumber) & "|" & UCase(componentType)
        End If

        If bomData.Exists(bomKey) Then

            itemData = bomData(bomKey)
            quantity = CLng(itemData(1)) + 1
            itemData(1) = quantity

            If CDbl(itemData(4)) > 0 Then
                itemData(5) = CDbl(itemData(4)) * quantity
            End If

            bomData(bomKey) = itemData

        Else

            quantity = 1
            totalMass = massEach

            ReDim itemData(0 To 16)

            itemData(0) = partNumber
            itemData(1) = quantity
            itemData(2) = componentType
            itemData(3) = materialName
            itemData(4) = massEach
            itemData(5) = totalMass
            itemData(6) = volumeMM3
            itemData(7) = density
            itemData(8) = thicknessMM
            itemData(9) = thicknessMethod
            itemData(10) = thicknessStatus
            itemData(11) = nomenclature
            itemData(12) = revision
            itemData(13) = description
            itemData(14) = instanceName
            itemData(15) = parentName
            itemData(16) = filePath

            bomData.Add bomKey, itemData

        End If

        If child.Products.Count > 0 Then
            Call ReadProductTree(child, bomData)
        End If

    Next i

End Sub


Private Sub GetPhysicalProperties( _
    ByVal productObj As Product, _
    ByRef massKg As Double, _
    ByRef volumeMM3 As Double, _
    ByRef densityKgM3 As Double)

    Dim refProd As Product
    Dim inertiaObj As Object
    Dim analyzeObj As Object
    Dim volumeM3 As Double

    massKg = 0#
    volumeMM3 = 0#
    densityKgM3 = 0#

    On Error Resume Next

    Set refProd = productObj.ReferenceProduct

    If Not refProd Is Nothing Then
        Set inertiaObj = refProd.GetTechnologicalObject("Inertia")
    End If

    If inertiaObj Is Nothing Then
        Set inertiaObj = productObj.GetTechnologicalObject("Inertia")
    End If

    If Not inertiaObj Is Nothing Then
        massKg = inertiaObj.Mass
        densityKgM3 = inertiaObj.Density
    End If

    Set analyzeObj = Nothing

    If Not refProd Is Nothing Then
        Set analyzeObj = refProd.Analyze
    End If

    If analyzeObj Is Nothing Then
        Set analyzeObj = productObj.Analyze
    End If

    If Not analyzeObj Is Nothing Then

        volumeM3 = analyzeObj.Volume

        If volumeM3 > 0# Then
            volumeMM3 = volumeM3 * 1000000000#
        End If

    End If

    On Error GoTo 0

End Sub


Private Sub GetThickness( _
    ByVal productObj As Product, _
    ByVal volumeMM3 As Double, _
    ByRef thicknessMM As Double, _
    ByRef thicknessMethod As String, _
    ByRef thicknessStatus As String)

    Dim definedThickness As Double
    Dim wetAreaMM2 As Double

    thicknessMM = 0#
    thicknessMethod = ""
    thicknessStatus = ""

    definedThickness = FindThicknessParameter(productObj)

    If definedThickness > 0# Then

        thicknessMM = definedThickness
        thicknessMethod = "Parameter"
        thicknessStatus = "Defined thickness parameter"

        Exit Sub

    End If

    wetAreaMM2 = GetWetAreaMM2(productObj)

    If volumeMM3 > 0# And wetAreaMM2 > 0# Then

        thicknessMM = (2# * volumeMM3) / wetAreaMM2
        thicknessMethod = "Geometric estimate (2V/A)"
        thicknessStatus = _
            "Approximate - verify for non-sheet geometry"

        Exit Sub

    End If

    thicknessMM = 0#
    thicknessMethod = "Unavailable"
    thicknessStatus = "Could not determine thickness"

End Sub


Private Function FindThicknessParameter( _
    ByVal productObj As Product) As Double

    Dim refProd As Product
    Dim parametersObj As Parameters
    Dim parameterObj As Parameter
    Dim i As Long
    Dim paramName As String
    Dim paramValueString As String
    Dim numericValue As Double

    FindThicknessParameter = 0#

    On Error Resume Next

    Set refProd = productObj.ReferenceProduct

    If refProd Is Nothing Then
        Set refProd = productObj
    End If

    Set parametersObj = refProd.Parameters

    If parametersObj Is Nothing Then
        On Error GoTo 0
        Exit Function
    End If

    For i = 1 To parametersObj.Count

        Set parameterObj = parametersObj.Item(i)

        paramName = LCase(parameterObj.Name)

        If InStr(paramName, "thickness") > 0 Or _
           InStr(paramName, "thick") > 0 Or _
           InStr(paramName, "epaisseur") > 0 Then

            Err.Clear

            paramValueString = parameterObj.ValueAsString

            If Err.Number = 0 Then

                numericValue = ExtractNumber(paramValueString)

                If numericValue > 0# Then

                    FindThicknessParameter = numericValue

                    On Error GoTo 0
                    Exit Function

                End If

            End If

        End If

    Next i

    On Error GoTo 0

End Function


Private Function GetWetAreaMM2( _
    ByVal productObj As Product) As Double

    Dim refProd As Product
    Dim analyzeObj As Object

    GetWetAreaMM2 = 0#

    On Error Resume Next

    Set refProd = productObj.ReferenceProduct

    If Not refProd Is Nothing Then
        Set analyzeObj = refProd.Analyze
    End If

    If analyzeObj Is Nothing Then
        Set analyzeObj = productObj.Analyze
    End If

    If Not analyzeObj Is Nothing Then
        GetWetAreaMM2 = analyzeObj.WetArea
    End If

    On Error GoTo 0

End Function


Private Function GetMaterialName( _
    ByVal productObj As Product) As String

    Dim refProd As Product
    Dim managerObj As Object
    Dim materialObj As Object
    Dim refDoc As Object
    Dim partObj As Object

    GetMaterialName = ""

    On Error Resume Next

    Set refProd = productObj.ReferenceProduct

    If Not refProd Is Nothing Then

        Set managerObj = refProd.GetItem("CATMatManagerVBExt")

        If Not managerObj Is Nothing Then

            Set materialObj = Nothing

            managerObj.GetMaterialOnProduct _
                refProd, materialObj

            If Not materialObj Is Nothing Then

                GetMaterialName = materialObj.Name

                On Error GoTo 0
                Exit Function

            End If

        End If

    End If

    Set refDoc = Nothing

    If Not refProd Is Nothing Then
        Set refDoc = refProd.Parent
    End If

    If Not refDoc Is Nothing Then

        Set partObj = Nothing
        Set partObj = refDoc.Part

        If Not partObj Is Nothing Then

            Set managerObj = Nothing
            Set managerObj = partObj.GetItem("CATMatManagerVBExt")

            If Not managerObj Is Nothing Then

                Set materialObj = Nothing

                managerObj.GetMaterialOnPart _
                    partObj, materialObj

                If Not materialObj Is Nothing Then
                    GetMaterialName = materialObj.Name
                End If

            End If

        End If

    End If

    On Error GoTo 0

End Function


Private Function GetProductFilePath( _
    ByVal productObj As Product) As String

    Dim refProd As Product
    Dim parentObj As Object

    GetProductFilePath = ""

    On Error Resume Next

    Set refProd = productObj.ReferenceProduct

    If Not refProd Is Nothing Then

        Set parentObj = refProd.Parent
        GetProductFilePath = parentObj.FullName

    End If

    On Error GoTo 0

End Function


Private Function SafePartNumber( _
    ByVal productObj As Product) As String

    SafePartNumber = ""

    On Error Resume Next

    SafePartNumber = productObj.PartNumber

    If Trim(SafePartNumber) = "" Then
        SafePartNumber = productObj.Name
    End If

    On Error GoTo 0

End Function


Private Function SafeInstanceName( _
    ByVal productObj As Product) As String

    SafeInstanceName = ""

    On Error Resume Next
    SafeInstanceName = productObj.Name
    On Error GoTo 0

End Function


Private Function SafeNomenclature( _
    ByVal productObj As Product) As String

    SafeNomenclature = ""

    On Error Resume Next
    SafeNomenclature = productObj.Nomenclature
    On Error GoTo 0

End Function


Private Function SafeRevision( _
    ByVal productObj As Product) As String

    SafeRevision = ""

    On Error Resume Next
    SafeRevision = productObj.Revision
    On Error GoTo 0

End Function


Private Function SafeDescription( _
    ByVal productObj As Product) As String

    SafeDescription = ""

    On Error Resume Next

    SafeDescription = productObj.DescriptionRef

    If SafeDescription = "" Then
        SafeDescription = productObj.DescriptionInst
    End If

    On Error GoTo 0

End Function


Private Function ExtractNumber( _
    ByVal valueString As String) As Double

    Dim cleaned As String

    cleaned = Trim(valueString)
    cleaned = Replace(cleaned, ",", ".")

    On Error Resume Next
    ExtractNumber = Val(cleaned)
    On Error GoTo 0

End Function


Private Function CSVLine( _
    ByVal fields As Variant) As String

    Dim i As Long
    Dim result As String

    result = ""

    For i = LBound(fields) To UBound(fields)

        If i > LBound(fields) Then
            result = result & ","
        End If

        result = result & CSVField(CStr(fields(i)))

    Next i

    CSVLine = result

End Function


Private Function CSVField( _
    ByVal value As String) As String

    Dim cleanedValue As String

    cleanedValue = Replace(value, , "")

    CSVField =  & cleanedValue & 

End Function


Private Function FormatCSVNumber( _
    ByVal value As Variant) As String

    If IsNumeric(value) Then

        If CDbl(value) = 0# Then
            FormatCSVNumber = ""
        Else
            FormatCSVNumber = Format(CDbl(value), "0.######")
        End If

    Else

        FormatCSVNumber = ""

    End If

End Function

Notes & Limitations

Mass and density depend on CATIA being able to evaluate the component's inertia properties. Material may be blank if no CATIA material is applied or if the model stores material information differently.

The geometric thickness estimate is not equivalent to manually using CATIA's interactive Measure Thickness command. Rows marked Geometric estimate (2V/A) should be treated as values requiring engineering verification.

Duplicate components are grouped primarily by referenced file path when available, reducing the chance of combining unrelated files that share the same Part Number.

Viewing the CSV

The report does not require Microsoft Excel. The CSV can be opened in Excel, LibreOffice Calc, a text editor, or the separate browser-based CATIA BOM CSV Viewer.
