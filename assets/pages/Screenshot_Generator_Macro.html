CATIA Screenshot Generator

Saves isometric, front, top and right-side images of the active document, or of every part in an assembly, for reports, work instructions and Power Apps records. Two code blocks to paste: the form and the module.

What it does

  • Two sources. The active document exactly as it is shown, or each part of the active product, opened in its own window one at a time.
  • Four views. Isometric, front, top and right side. Tick the ones you want. Each view is fitted to the screen before the capture.
  • White background and no tree if you tick those boxes. The screen is put back the way it was afterwards.
  • File names come from the part number: BRKT-001_ISO.jpg, BRKT-001_FRONT.jpg and so on.
  • JPG, BMP or TIFF.
  • English / Korean button in the top right corner. The language is remembered for next time.
  • Credit image centered at the bottom, stored inside the module as text, the same as your other macros.
  • Works beside your other macros. Every shared name starts with Shot, so it does not clash with the other macros in the same VBA project.

Setup

  1. In the VBA editor, use Insert, UserForm. In the Properties window set its (Name) to frmScreenshots. Leave the form empty in the designer.
  2. Open the form's code (right-click the form, View Code), select all, delete, and paste the Form code below.
  3. Use Insert, Module. In the Properties window set its (Name) to ScreenshotGenerator, then paste the Module code below.
  4. Use Debug, Compile VBAProject.
  5. Open a CATProduct or CATPart, then run CATMain from the ScreenshotGenerator module.

First run: this code was written and checked for structure, but not run inside CATIA. Try it on one part first and check that the front, top and right views match what your team calls front, top and right.

Private Sub DoAction(ByVal actionId As Long)

    Dim message As String
    Dim ok As Boolean

    If mBusy Then Exit Sub

    If Not DocReady() Then
        ShowDocStatus
        Exit Sub
    End If

    SaveInputs

    SetBusy True

    ShowStatus ShotTxt("stWorking")

    On Error GoTo Failed

    ok = ShotRun(Me, actionId, message)

    On Error Resume Next

    CATIA.RefreshDisplay = True
    CATIA.DisplayFileAlerts = True

    SetBusy False

    ShowStatus message

    Exit Sub

Failed:

    message = Err.Description

    On Error Resume Next

    CATIA.RefreshDisplay = True
    CATIA.DisplayFileAlerts = True

    SetBusy False

    ShowStatus ShotFmt(ShotTxt("msgError"), message)

End Sub

Module code

ScreenshotGenerator

The long block of letters and numbers at the bottom is the credit image. Do not edit it.

Option Explicit

'=========================================================
' CATIA SCREENSHOT GENERATOR - MAIN MODULE
' Module name: ScreenshotGenerator
' Run CATMain to open the window.
'=========================================================

Private Const REG_APP As String = "CatiaScreenshotGenerator"
Private Const REG_SEC As String = "Settings"

' "EN" or "KO"
Public gShotLang As String


'=========================================================
' ENTRY POINT
'=========================================================

Sub CATMain()

    ShotLoadLanguage
    frmScreenshots.Show

End Sub

'=========================================================
' CAPTURE
'=========================================================

Public Function ShotRun( _
        ByVal frm As Object, _
        ByVal actionId As Long, _
        ByRef message As String) As Boolean

    Dim docs As Object
    Dim keys As Variant
    Dim d As Object
    Dim homeWindow As Object
    Dim outFolder As String
    Dim ext As String
    Dim fmt As Long
    Dim i As Long
    Dim made As Long
    Dim shots As Long
    Dim skipped As Long
    Dim before As Long
    Dim opened As Boolean

    ShotRun = False

    outFolder = WithSlash(CtlText(frm, "txtOut"))

    If Not FolderExists(outFolder) Then
        message = ShotTxt("msgNoOut")
        Exit Function
    End If

    If Not (CtlOn(frm, "chkIso") Or CtlOn(frm, "chkFront") Or CtlOn(frm, "chkTop") Or CtlOn(frm, "chkRight")) Then
        message = ShotTxt("msgNoView")
        Exit Function
    End If

    ' CATIA capture formats: 2 = TIFF, 4 = BMP, 5 = JPEG
    Select Case CtlIndex(frm, "cboFormat")
        Case 1: ext = "bmp": fmt = 4
        Case 2: ext = "tif": fmt = 2
        Case Else: ext = "jpg": fmt = 5
    End Select

    Set d = CATIA.ActiveDocument
    Set homeWindow = CATIA.ActiveWindow

    If CtlIndex(frm, "cboSource") = 1 And TypeName(d) = "ProductDocument" Then

        '---------------------------------------
        ' Each part in its own window
        '---------------------------------------

        On Error Resume Next
        d.Product.ApplyWorkMode 2
        On Error GoTo 0

        Set docs = CreateObject("Scripting.Dictionary")

        CollectDocs d.Product, docs, True, False

        If docs.Count = 0 Then
            message = ShotTxt("msgNothing")
            Exit Function
        End If

        keys = docs.Keys

        For i = 0 To docs.Count - 1

            Set d = docs.Item(keys(i))

            Say frm, ShotFmt(ShotTxt("stItem"), CStr(i + 1), CStr(docs.Count)) & " " & d.Name

            made = 0

            If Len(d.Path) > 0 Then

                opened = False

                On Error Resume Next
                before = CATIA.Windows.Count
                CATIA.Documents.Open d.FullName
                opened = (CATIA.Windows.Count > before)
                On Error GoTo 0

                made = ShootDoc(d, frm, outFolder, ext, fmt)

                On Error Resume Next
                If opened Then CATIA.ActiveWindow.Close
                On Error GoTo 0

            End If

            If made > 0 Then
                shots = shots + made
            Else
                skipped = skipped + 1
            End If

        Next i

        On Error Resume Next
        homeWindow.Activate
        On Error GoTo 0

    Else

        shots = ShootDoc(d, frm, outFolder, ext, fmt)

    End If

    message = ShotFmt(ShotTxt("msgDone"), CStr(shots), CStr(skipped))

    ShotRun = True

End Function


' Captures the ticked views of the document shown in the active
' window. Returns the number of images saved.
Private Function ShootDoc( _
        ByVal d As Object, _
        ByVal frm As Object, _
        ByVal outFolder As String, _
        ByVal ext As String, _
        ByVal fmt As Long) As Long

    Dim win As Object
    Dim viewer As Object
    Dim vp As Object
    Dim oldColor(2) As Variant
    Dim white(2) As Variant
    Dim oldSight(2) As Variant
    Dim oldUp(2) As Variant
    Dim oldLayout As Long
    Dim baseName As String
    Dim made As Long

    On Error GoTo Failed

    Set win = CATIA.ActiveWindow
    Set viewer = win.ActiveViewer
    Set vp = viewer.Viewpoint3D

    On Error Resume Next
    baseName = d.Product.PartNumber
    On Error GoTo Failed

    If Len(baseName) = 0 Then baseName = StripExt(d.Name)

    baseName = outFolder & CleanName(baseName)

    ' Remember the screen as it was
    oldLayout = win.Layout
    viewer.GetBackgroundColor oldColor
    vp.GetSightDirection oldSight
    vp.GetUpDirection oldUp

    ' Layout 1 = geometry only, no specification tree
    If CtlOn(frm, "chkTree") Then win.Layout = 1

    If CtlOn(frm, "chkWhite") Then
        white(0) = 1
        white(1) = 1
        white(2) = 1
        viewer.PutBackgroundColor white
    End If

    ' Sight direction (where the camera looks) and up direction
    If CtlOn(frm, "chkIso") Then made = made + Shoot(viewer, vp, -1, -1, -1, 0, 0, 1, baseName & "_ISO." & ext, fmt)
    If CtlOn(frm, "chkFront") Then made = made + Shoot(viewer, vp, -1, 0, 0, 0, 0, 1, baseName & "_FRONT." & ext, fmt)
    If CtlOn(frm, "chkTop") Then made = made + Shoot(viewer, vp, 0, 0, -1, 0, 1, 0, baseName & "_TOP." & ext, fmt)
    If CtlOn(frm, "chkRight") Then made = made + Shoot(viewer, vp, 0, -1, 0, 0, 0, 1, baseName & "_RIGHT." & ext, fmt)

    ' Put the screen back
    On Error Resume Next

    viewer.PutBackgroundColor oldColor
    vp.PutSightDirection oldSight
    vp.PutUpDirection oldUp
    win.Layout = oldLayout
    viewer.Reframe
    viewer.Update

    ShootDoc = made

    Exit Function

Failed:

    ShootDoc = made

    On Error Resume Next

    viewer.PutBackgroundColor oldColor
    win.Layout = oldLayout

End Function


' Turns the camera, fits the model and saves one image.
' Returns 1 when the image was saved, otherwise 0.
Private Function Shoot( _
        ByVal viewer As Object, _
        ByVal vp As Object, _
        ByVal sx As Double, _
        ByVal sy As Double, _
        ByVal sz As Double, _
        ByVal ux As Double, _
        ByVal uy As Double, _
        ByVal uz As Double, _
        ByVal filePath As String, _
        ByVal fmt As Long) As Long

    Dim sight(2) As Variant
    Dim up(2) As Variant

    On Error GoTo Failed

    sight(0) = sx
    sight(1) = sy
    sight(2) = sz

    up(0) = ux
    up(1) = uy
    up(2) = uz

    vp.PutSightDirection sight
    vp.PutUpDirection up

    viewer.Reframe
    viewer.Update

    DeleteFile filePath

    viewer.CaptureToFile fmt, filePath

    If FileExists(filePath) Then Shoot = 1

    Exit Function

Failed:

End Function

' "" = nothing open, otherwise Product, Part, Drawing or Other
Public Function ShotDocKind() As String

    Dim t As String

    On Error Resume Next

    If CATIA.Documents.Count > 0 Then t = TypeName(CATIA.ActiveDocument)

    On Error GoTo 0

    Select Case t
        Case "ProductDocument": ShotDocKind = "Product"
        Case "PartDocument": ShotDocKind = "Part"
        Case "DrawingDocument": ShotDocKind = "Drawing"
        Case "": ShotDocKind = ""
        Case Else: ShotDocKind = "Other"
    End Select

End Function


Public Function ShotDocName() As String

    On Error Resume Next

    ShotDocName = CATIA.ActiveDocument.Name

End Function

' Replaces characters that are not allowed in a file name
Private Function CleanName(ByVal rawName As String) As String

    Dim i As Long
    Dim ch As String
    Dim code As Long
    Dim result As String

    For i = 1 To Len(rawName)

        ch = Mid$(rawName, i, 1)
        code = AscW(ch)

        If InStr("\/:*?""<>|", ch) > 0 Or (code >= 0 And code < 32) Then
            result = result & "_"
        Else
            result = result & ch
        End If

    Next i

    If Len(result) = 0 Then result = "Unnamed"

    CleanName = result

End Function


Private Function StripExt(ByVal fileName As String) As String

    Dim p As Long

    p = InStrRev(fileName, ".")

    If p > 1 Then
        StripExt = Left$(fileName, p - 1)

        StripExt = fileName
    End If

End Function

' Saves text as UTF-8 so Excel shows Korean correctly
Private Function WriteTextFile(ByVal filePath As String, ByVal content As String) As Boolean

    Dim stm As Object

' Splits one CSV line into its fields
Private Function CsvSplit(ByVal lineText As String) As Collection

    Dim fields As New Collection
    Dim i As Long
    Dim ch As String
    Dim cur As String
    Dim inQuotes As Boolean

    i = 1

    Do While i <= Len(lineText)

        ch = Mid$(lineText, i, 1)

        If inQuotes Then
            If ch = """" Then
                If Mid$(lineText, i + 1, 1) = """" Then
                    cur = cur & """"
                    i = i + 1
                Else
                    inQuotes = False
                End If
            Else
                cur = cur & ch
            End If
        ElseIf ch = """" Then
            inQuotes = True
        ElseIf ch = "," Then
            fields.Add cur
            cur = ""
        Else
            cur = cur & ch
        End If

        i = i + 1

    Loop

    fields.Add cur

    Set CsvSplit = fields

End Function

Private Function DocOf(ByVal prod As Object) As Object

    On Error Resume Next

    Set d = prod.ReferenceProduct.Parent

    If d Is Nothing Then Exit Function

    If Right$(TypeName(d), 8) = "Document" Then Set DocOf = d

End Function


Private Sub AddDoc(ByVal docs As Object, ByVal d As Object)

    Dim key As String

    key = UCase$(d.FullName)

    If Not docs.Exists(key) Then docs.Add key, d

End Sub


' Fills docs (a Dictionary) with every loaded document under prod
Private Sub CollectDocs( _
        ByVal prod As Object, _
        ByVal docs As Object, _
        ByVal wantParts As Boolean, _
        ByVal wantProducts As Boolean)

    Dim kids As Object
    Dim child As Object
    Dim d As Object
    Dim kind As String
    Dim i As Long

    On Error Resume Next
    Set kids = prod.Products
    On Error GoTo 0

    If kids Is Nothing Then Exit Sub

    For i = 1 To kids.Count

        Set child = kids.Item(i)
        Set d = DocOf(child)

        If Not d Is Nothing Then

            kind = TypeName(d)

            If kind = "PartDocument" And wantParts Then AddDoc docs, d
            If kind = "ProductDocument" And wantProducts Then AddDoc docs, d

            CollectDocs child, docs, wantParts, wantProducts

        End If

    Next i

End Sub

' Turns \uXXXX codes into real characters
Private Function U(ByVal coded As String) As String

    Dim i As Long
    Dim result As String

    i = 1

    Do While i <= Len(coded)

        If Mid$(coded, i, 2) = "\u" And i + 5 <= Len(coded) Then
            result = result & ChrW$(CLng("&H" & Mid$(coded, i + 2, 4)))
            i = i + 6
        Else
            result = result & Mid$(coded, i, 1)
            i = i + 1
        End If

    Loop

    U = result

End Function

' LOGO
' The logo picture is stored below as text.
' The form shows it with:
'     ShotLoadLogo Me.Controls("imgLogo")

' Puts the logo into an Image control.
' Returns False if the picture could not be loaded.
Public Function ShotLoadLogo(ByVal target As Object) As Boolean

    Dim data() As Byte
    Dim filePath As String
    Dim fileNo As Integer

    ShotLoadLogo = False

    On Error GoTo Failed

    data = HexToBytes(LogoHex())

    ' VBA can only load a picture from a file, so the data is
    ' written to the Windows temp folder, loaded, then deleted.
    filePath = Environ$("TEMP") & "\catia_macro_logo.gif"

    If Len(Dir$(filePath)) > 0 Then Kill filePath

    fileNo = FreeFile

    Open filePath For Binary Access Write As #fileNo
    Put #fileNo, , data
    Close #fileNo

    Set target.Picture = LoadPicture(filePath)

    ShotLoadLogo = True

    On Error Resume Next
    Kill filePath

    Exit Function

Failed:

    On Error Resume Next
    Close #fileNo

End Function


' Turns pairs of hex characters back into bytes
Private Function HexToBytes(ByVal hexText As String) As Byte()

    Dim count As Long
    Dim i As Long
    Dim data() As Byte

    count = Len(hexText) \ 2

    ReDim data(0 To count - 1)

    For i = 0 To count - 1
        data(i) = CByte("&H" & Mid$(hexText, i * 2 + 1, 2))
    Next i

    HexToBytes = data

End Function

Known limitations

  • I said PNG when I suggested this macro. CATIA's own capture command does not offer PNG in R21, so the choices are JPG, BMP and TIFF.
  • The image size is the size of the CATIA window. Maximize CATIA for larger images.
  • The view directions follow the part axes: front looks along X, top looks down Z, right looks along Y. If your parts are modeled in another orientation, change the six numbers in the four Shoot lines of ShootDoc.
  • The compass and the small axis symbol in the corner are not hidden. Turn them off in the View menu before running if you do not want them in the images.
  • Parts that were never saved are skipped in the each-part mode, because they cannot be opened in their own window.
  • A white background works best with the graduated background option turned off in Tools, Options, General, Display, Visualization.
  • The window is modal, so CATIA cannot be clicked while it is open. Anything that must be selected in CATIA has to be selected before running the macro.
  • If the credit image cannot be loaded, for example when the Windows temp folder is not writable, the window opens without it.
