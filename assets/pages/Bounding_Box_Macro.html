CATIA Bounding Box Calculator

Measures the overall size of the active part, or of every part in an assembly, and writes length, width and height into each part as parameters and as a "Stock Size" property that can be added to the bill of material. Two code blocks to paste: the form and the module.

What it does

  • One part or the whole assembly. Each part is measured once.
  • Largest size first. The three sizes are sorted, so Length is always the longest. An optional allowance in mm is added to each one.
  • Written into the part. Parameters BBx_Length, BBx_Width and BBx_Height, and a property called Stock Size such as 120.0 x 80.0 x 25.0. Running again updates them instead of adding copies.
  • CSV list with the raw sizes along X, Y and Z and the sorted stock size, saved as DocumentName_StockSize.csv.
  • Leaves the part clean. The six temporary points used for the measurement are deleted afterwards.
  • English / Korean button in the top right corner. The language is remembered for next time.
  • Credit image centered at the bottom, stored inside the module as text, the same as your other macros.
  • Works beside your other macros. Every shared name starts with BBx, so it does not clash with the other macros in the same VBA project.

Setup

  1. In the VBA editor, use Insert, UserForm. In the Properties window set its (Name) to frmBoundingBox. Leave the form empty in the designer.
  2. Open the form's code (right-click the form, View Code), select all, delete, and paste the Form code below.
  3. Use Insert, Module. In the Properties window set its (Name) to BoundingBox, then paste the Module code below.
  4. Use Debug, Compile VBAProject.
  5. Open a CATPart or CATProduct, then run CATMain from the BoundingBox module.

First run: this code was written and checked for structure, but not run inside CATIA. Try it on a simple block of known size first, and check the three numbers and that no "BBx_Temp" set is left in the tree.

Private Sub DoAction(ByVal actionId As Long)

    Dim message As String
    Dim ok As Boolean

    If mBusy Then Exit Sub

    If Not DocReady() Then
        ShowDocStatus
        Exit Sub
    End If

    SaveInputs

    SetBusy True

    ShowStatus BBxTxt("stWorking")

    On Error GoTo Failed

    ok = BBxRun(Me, actionId, message)

    On Error Resume Next

    CATIA.RefreshDisplay = True
    CATIA.DisplayFileAlerts = True

    SetBusy False

    ShowStatus message

    Exit Sub

Failed:

    message = Err.Description

    On Error Resume Next

    CATIA.RefreshDisplay = True
    CATIA.DisplayFileAlerts = True

    SetBusy False

    ShowStatus BBxFmt(BBxTxt("msgError"), message)

End Sub

Module code

The long block of letters and numbers at the bottom is the credit image. Do not edit it.

Option Explicit

'=========================================================
' CATIA BOUNDING BOX CALCULATOR - MAIN MODULE
' Module name: BoundingBox
' Run CATMain to open the window.
'=========================================================

Private Const REG_APP As String = "CatiaBoundingBox"
Private Const REG_SEC As String = "Settings"

' "EN" or "KO"
Public gBBxLang As String

' Names written into each part
Private Const PARAM_L As String = "BBx_Length"
Private Const PARAM_W As String = "BBx_Width"
Private Const PARAM_H As String = "BBx_Height"
Private Const PROP_STOCK As String = "Stock Size"
Private Const TEMP_SET As String = "BBx_Temp"


'=========================================================
' ENTRY POINT
'=========================================================

Sub CATMain()

    BBxLoadLanguage
    frmBoundingBox.Show

End Sub

'=========================================================
' MEASURE
'=========================================================

Public Function BBxRun( _
        ByVal frm As Object, _
        ByVal actionId As Long, _
        ByRef message As String) As Boolean

    Dim docs As Object
    Dim keys As Variant
    Dim d As Object
    Dim outFolder As String
    Dim target As String
    Dim content As String
    Dim allowText As String
    Dim stock As String
    Dim errText As String
    Dim partNo As String
    Dim allowance As Double
    Dim sx As Double
    Dim sy As Double
    Dim sz As Double
    Dim boxL As Double
    Dim boxW As Double
    Dim boxH As Double
    Dim i As Long
    Dim done As Long
    Dim failed As Long
    Dim wantCsv As Boolean

    BBxRun = False

    wantCsv = CtlOn(frm, "chkCsv")
    outFolder = WithSlash(CtlText(frm, "txtOut"))

    If wantCsv And Not FolderExists(outFolder) Then
        message = BBxTxt("msgNoOut")
        Exit Function
    End If

    allowText = Replace(CtlText(frm, "txtAllow"), ",", ".")

    If Len(allowText) = 0 Then allowText = "0"

    If Not IsNumeric(allowText) Then
        message = BBxTxt("msgBadAllow")
        Exit Function
    End If

    allowance = Val(allowText)

    If allowance < 0 Then
        message = BBxTxt("msgBadAllow")
        Exit Function
    End If

    Set d = CATIA.ActiveDocument
    Set docs = CreateObject("Scripting.Dictionary")

    If TypeName(d) = "PartDocument" Then

        AddDoc docs, d

    ElseIf CtlIndex(frm, "cboScope") = 1 Then

        On Error Resume Next
        d.Product.ApplyWorkMode 2
        On Error GoTo 0

        CollectDocs d.Product, docs, True, False

    Else

        message = BBxTxt("msgNeedPart")
        Exit Function

    End If

    If docs.Count = 0 Then
        message = BBxTxt("msgNothing")
        Exit Function
    End If

    content = Csv(BBxTxt("hPN")) & "," & Csv(BBxTxt("hFile")) & "," & Csv(BBxTxt("hX")) & "," & _
              Csv(BBxTxt("hY")) & "," & Csv(BBxTxt("hZ")) & "," & Csv(BBxTxt("hL")) & "," & _
              Csv(BBxTxt("hW")) & "," & Csv(BBxTxt("hH")) & "," & Csv(BBxTxt("hStock")) & "," & _
              Csv(BBxTxt("hNote")) & vbCrLf

    keys = docs.Keys

    For i = 0 To docs.Count - 1

        Set d = docs.Item(keys(i))

        partNo = ""

        On Error Resume Next
        partNo = d.Product.PartNumber
        On Error GoTo 0

        Say frm, BBxFmt(BBxTxt("stItem"), CStr(i + 1), CStr(docs.Count)) & " " & partNo

        errText = ""

        If MeasureBox(d, sx, sy, sz, errText) Then

            ' Largest size first, then the allowance on every side
            SortThree sx, sy, sz, boxL, boxW, boxH

            boxL = boxL + allowance
            boxW = boxW + allowance
            boxH = boxH + allowance

            stock = Num(boxL, 1) & " x " & Num(boxW, 1) & " x " & Num(boxH, 1)

            If CtlOn(frm, "chkParam") Then
                SetLength d, PARAM_L, boxL
                SetLength d, PARAM_W, boxW
                SetLength d, PARAM_H, boxH
            End If

            If CtlOn(frm, "chkProp") Then SetStockProperty d, stock

            content = content & Csv(partNo) & "," & Csv(d.Name) & "," & Num(sx, 3) & "," & Num(sy, 3) & "," & _
                      Num(sz, 3) & "," & Num(boxL, 3) & "," & Num(boxW, 3) & "," & Num(boxH, 3) & "," & _
                      Csv(stock) & "," & vbCrLf

            done = done + 1

        Else

            content = content & Csv(partNo) & "," & Csv(d.Name) & ",,,,,,,," & Csv(errText) & vbCrLf

            failed = failed + 1

        End If

    Next i

    If wantCsv Then

        target = outFolder & CleanName(StripExt(BBxDocName())) & "_StockSize.csv"

        If Not WriteTextFile(target, content) Then
            message = BBxTxt("msgWriteFail")
            Exit Function
        End If

        OpenFile target

    End If

    If docs.Count = 1 And done = 1 Then
        message = BBxFmt(BBxTxt("msgOne"), partNo, stock)
    Else
        message = BBxFmt(BBxTxt("msgDone"), CStr(done), CStr(failed))
    End If

    BBxRun = True

End Function


' Puts three sizes in order, largest first
Private Sub SortThree( _
        ByVal a As Double, _
        ByVal b As Double, _
        ByVal c As Double, _
        ByRef big As Double, _
        ByRef middle As Double, _
        ByRef small As Double)

    Dim t As Double

    If a < b Then
        t = a
        a = b
        b = t
    End If

    If a < c Then
        t = a
        a = c
        c = t
    End If

    If b < c Then
        t = b
        b = c
        c = t
    End If

    big = a
    middle = b
    small = c

End Sub


'=========================================================
' BOUNDING BOX OF ONE PART
' Six temporary extremum points are created on the main
' body, measured and removed again. The sizes are along
' the X, Y and Z axes of the part.
'=========================================================

Private Function MeasureBox( _
        ByVal d As Object, _
        ByRef sx As Double, _
        ByRef sy As Double, _
        ByRef sz As Double, _
        ByRef errText As String) As Boolean

    Dim prt As Object
    Dim hsf As Object
    Dim mainBody As Object
    Dim tempSet As Object
    Dim spa As Object
    Dim bodyRef As Object
    Dim ext As Object
    Dim meas As Object
    Dim keepWork As Object
    Dim dirs(2) As Object
    Dim lo(2) As Double
    Dim hi(2) As Double
    Dim pt(2) As Variant
    Dim axis As Long
    Dim side As Long
    Dim other1 As Long
    Dim other2 As Long

    MeasureBox = False

    On Error GoTo Failed

    Set prt = d.Part
    Set mainBody = prt.MainBody

    If mainBody.Shapes.Count = 0 Then
        errText = BBxTxt("dEmpty")
        Exit Function
    End If

    Set hsf = prt.HybridShapeFactory
    Set spa = d.GetWorkbench("SPAWorkbench")
    Set bodyRef = prt.CreateReferenceFromObject(mainBody)

    On Error Resume Next
    Set keepWork = prt.InWorkObject
    On Error GoTo Failed

    Set dirs(0) = hsf.AddNewDirectionByCoord(1, 0, 0)
    Set dirs(1) = hsf.AddNewDirectionByCoord(0, 1, 0)
    Set dirs(2) = hsf.AddNewDirectionByCoord(0, 0, 1)

    Set tempSet = prt.HybridBodies.Add

    On Error Resume Next
    tempSet.Name = TEMP_SET
    On Error GoTo Failed

    For axis = 0 To 2

        other1 = (axis + 1) Mod 3
        other2 = (axis + 2) Mod 3

        ' side 0 = lowest point, side 1 = highest point
        For side = 0 To 1

            Set ext = hsf.AddNewExtremum(bodyRef, dirs(axis), side)

            ' Two more directions so the result is always one point
            ext.Direction2 = dirs(other1)
            ext.ExtremumType2 = 1
            ext.Direction3 = dirs(other2)
            ext.ExtremumType3 = 1

            tempSet.AppendHybridShape ext

            prt.UpdateObject ext

            Set meas = spa.GetMeasurable(prt.CreateReferenceFromObject(ext))

            meas.GetPoint pt

            If side = 0 Then
                lo(axis) = CDbl(pt(axis))
            Else
                hi(axis) = CDbl(pt(axis))
            End If

        Next side

    Next axis

    sx = hi(0) - lo(0)
    sy = hi(1) - lo(1)
    sz = hi(2) - lo(2)

    RemoveTemp prt, hsf, tempSet, keepWork

    MeasureBox = True

    Exit Function

Failed:

    errText = Err.Description

    RemoveTemp prt, hsf, tempSet, keepWork

End Function


' Deletes the temporary geometrical set and puts the
' in-work object back
Private Sub RemoveTemp( _
        ByVal prt As Object, _
        ByVal hsf As Object, _
        ByVal tempSet As Object, _
        ByVal keepWork As Object)

    On Error Resume Next

    If Not tempSet Is Nothing Then
        hsf.DeleteObjectForDatum prt.CreateReferenceFromObject(tempSet)
    End If

    If Not keepWork Is Nothing Then prt.InWorkObject = keepWork

    prt.Update

End Sub

' Creates the length parameter, or changes it when it exists
Private Sub SetLength(ByVal d As Object, ByVal paramName As String, ByVal sizeMm As Double)

    Dim params As Object
    Dim direct As Object
    Dim p As Object
    Dim i As Long

    On Error Resume Next

    Set params = d.Part.Parameters
    Set direct = params.RootParameterSet.DirectParameters

    If Not direct Is Nothing Then

        For i = 1 To direct.Count

            Set p = direct.Item(i)

            If StrComp(ShortName(p.Name), paramName, vbTextCompare) = 0 Then
                p.Value = sizeMm
                Exit Sub
            End If

        Next i

    End If

    params.CreateDimension paramName, "LENGTH", sizeMm

End Sub


' Creates the Stock Size property, or changes it when it exists
Private Sub SetStockProperty(ByVal d As Object, ByVal stock As String)

    Dim props As Object
    Dim p As Object
    Dim i As Long

    On Error Resume Next

    Set props = d.Product.UserRefProperties

    If props Is Nothing Then Exit Sub

    For i = 1 To props.Count

        Set p = props.Item(i)

        If StrComp(ShortName(p.Name), PROP_STOCK, vbTextCompare) = 0 Then
            p.ValuateFromString stock
            Exit Sub
        End If

    Next i

    props.CreateString PROP_STOCK, stock

End Sub

' "" = nothing open, otherwise Product, Part, Drawing or Other
Public Function BBxDocKind() As String

    Dim t As String

    On Error Resume Next

    If CATIA.Documents.Count > 0 Then t = TypeName(CATIA.ActiveDocument)

    On Error GoTo 0

    Select Case t
        Case "ProductDocument": BBxDocKind = "Product"
        Case "PartDocument": BBxDocKind = "Part"
        Case "DrawingDocument": BBxDocKind = "Drawing"
        Case "": BBxDocKind = ""
        Case Else: BBxDocKind = "Other"
    End Select

End Function

' Replaces characters that are not allowed in a file name
Private Function CleanName(ByVal rawName As String) As String

    Dim i As Long
    Dim ch As String
    Dim code As Long
    Dim result As String

    For i = 1 To Len(rawName)

        ch = Mid$(rawName, i, 1)
        code = AscW(ch)

        If InStr("\/:*?""<>|", ch) > 0 Or (code >= 0 And code < 32) Then
            result = result & "_"
        Else
            result = result & ch
        End If

    Next i

    If Len(result) = 0 Then result = "Unnamed"

    CleanName = result

End Function


Private Function StripExt(ByVal fileName As String) As String

    Dim p As Long

        StripExt = Left$(fileName, p - 1)
    Else
        StripExt = fileName
    End If

End Function

' A number written with a decimal point on every PC
Private Function Num(ByVal numberValue As Variant, Optional ByVal decimals As Long = 3) As String

    Dim pattern As String

    On Error Resume Next

    pattern = "0"

    If decimals > 0 Then pattern = "0." & String$(decimals, "0")

    Num = Replace(Format$(CDbl(numberValue), pattern), ",", ".")

' Splits one CSV line into its fields
Private Function CsvSplit(ByVal lineText As String) As Collection

    Dim fields As New Collection
    Dim i As Long
    Dim ch As String
    Dim cur As String
    Dim inQuotes As Boolean

    i = 1

    Do While i <= Len(lineText)

        ch = Mid$(lineText, i, 1)

        If inQuotes Then
            If ch = """" Then
                If Mid$(lineText, i + 1, 1) = """" Then
                    cur = cur & """"
                    i = i + 1
                Else
                    inQuotes = False
                End If
            Else
                cur = cur & ch
            End If
        ElseIf ch = """" Then
            inQuotes = True
        ElseIf ch = "," Then
            fields.Add cur
            cur = ""
        Else
            cur = cur & ch
        End If

        i = i + 1

    Loop

    fields.Add cur

    Set CsvSplit = fields

End Function

' Fills docs (a Dictionary) with every loaded document under prod
Private Sub CollectDocs( _
        ByVal prod As Object, _
        ByVal docs As Object, _
        ByVal wantParts As Boolean, _
        ByVal wantProducts As Boolean)

    Dim kids As Object
    Dim child As Object
    Dim d As Object
    Dim kind As String
    Dim i As Long

    On Error Resume Next
    Set kids = prod.Products
    On Error GoTo 0

    If kids Is Nothing Then Exit Sub

    For i = 1 To kids.Count

        Set child = kids.Item(i)
        Set d = DocOf(child)

        If Not d Is Nothing Then

            kind = TypeName(d)

            If kind = "PartDocument" And wantParts Then AddDoc docs, d
            If kind = "ProductDocument" And wantProducts Then AddDoc docs, d

            CollectDocs child, docs, wantParts, wantProducts

        End If

    Next i

End Sub

' Turns \uXXXX codes into real characters
Private Function U(ByVal coded As String) As String

    Dim i As Long
    Dim result As String

    i = 1

    Do While i <= Len(coded)

        If Mid$(coded, i, 2) = "\u" And i + 5 <= Len(coded) Then
            result = result & ChrW$(CLng("&H" & Mid$(coded, i + 2, 4)))
            i = i + 6
        Else
            result = result & Mid$(coded, i, 1)
            i = i + 1
        End If

    Loop

    U = result

End Function

' Puts the logo into an Image control.
' Returns False if the picture could not be loaded.
Public Function BBxLoadLogo(ByVal target As Object) As Boolean

    Dim data() As Byte
    Dim filePath As String
    Dim fileNo As Integer

    BBxLoadLogo = False

    On Error GoTo Failed

    data = HexToBytes(LogoHex())

    ' VBA can only load a picture from a file, so the data is
    ' written to the Windows temp folder, loaded, then deleted.
    filePath = Environ$("TEMP") & "\catia_macro_logo.gif"

    If Len(Dir$(filePath)) > 0 Then Kill filePath

    fileNo = FreeFile

    Open filePath For Binary Access Write As #fileNo
    Put #fileNo, , data
    Close #fileNo

    Set target.Picture = LoadPicture(filePath)

    BBxLoadLogo = True

    On Error Resume Next
    Kill filePath

    Exit Function

Failed:

    On Error Resume Next
    Close #fileNo

End Function


' Turns pairs of hex characters back into bytes
Private Function HexToBytes(ByVal hexText As String) As Byte()

    Dim count As Long
    Dim i As Long
    Dim data() As Byte

    count = Len(hexText) \ 2

    ReDim data(0 To count - 1)

    For i = 0 To count - 1
        data(i) = CByte("&H" & Mid$(hexText, i * 2 + 1, 2))
    Next i

    HexToBytes = data

End Function

Known limitations

  • The box is aligned with the X, Y and Z axes of each part. A part modeled at an angle to its own axes gets a larger box than the tightest possible one.
  • Only the main body (PartBody) is measured. Other bodies and surfaces are ignored.
  • The measurement uses the Generative Shape Design extremum command, so it needs a license that includes wireframe and surface tools.
  • The parts are modified (new parameters and a property), so CATIA will ask to save them.
  • If a measurement fails part way, a geometrical set named "BBx_Temp" can be left in that part. It is safe to delete.
  • To show the stock size in a CATIA bill of material, add the "Stock Size" property in Analyze, Bill of Material, Define formats.
  • The window is modal, so CATIA cannot be clicked while it is open. Anything that must be selected in CATIA has to be selected before running the macro.
  • If the credit image cannot be loaded, for example when the Windows temp folder is not writable, the window opens without it.
Prepared for CATIA V5 R21 and later, VBA. Measures part bounding boxes and writes stock sizes.
