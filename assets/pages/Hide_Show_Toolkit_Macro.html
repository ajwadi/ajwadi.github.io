CATIA Hide Show Toolkit

One click to hide, or show again, all planes, axis systems, sketches, points, lines, geometrical sets and assembly constraints in the active part or assembly. Two code blocks to paste: the form and the module.

What it does

  • Tick the types, then click Hide or Show. Planes, axis systems and sketches are ticked by default.
  • Works through the whole assembly. Every part under the active product is included.
  • Show brings them back. The same ticks with the Show button reverse the Hide.
  • Your choices are remembered for next time.
  • Count on the status line of how many elements were changed.
  • English / Korean button in the top right corner. The language is remembered for next time.
  • Credit image centered at the bottom, stored inside the module as text, the same as your other macros.
  • Works beside your other macros. Every shared name starts with Vis, so it does not clash with the other macros in the same VBA project.

Setup

  1. In the VBA editor, use Insert, UserForm. In the Properties window set its (Name) to frmHideShow. Leave the form empty in the designer.
  2. Open the form's code (right-click the form, View Code), select all, delete, and paste the Form code below.
  3. Use Insert, Module. In the Properties window set its (Name) to HideShowToolkit, then paste the Module code below.
  4. Use Debug, Compile VBAProject.
  5. Open a CATProduct or CATPart, then run CATMain from the HideShowToolkit module.

First run: this code was written and checked for structure, but not run inside CATIA. Try Hide and then Show on a small assembly first.

Private Sub DoAction(ByVal actionId As Long)

    Dim message As String
    Dim ok As Boolean

    If mBusy Then Exit Sub

    If Not DocReady() Then
        ShowDocStatus
        Exit Sub
    End If

    SaveInputs

    SetBusy True

    ShowStatus VisTxt("stWorking")

    On Error GoTo Failed

    ok = VisRun(Me, actionId, message)

    On Error Resume Next

    CATIA.RefreshDisplay = True
    CATIA.DisplayFileAlerts = True

    SetBusy False

    ShowStatus message

    Exit Sub

Failed:

    message = Err.Description

    On Error Resume Next

    CATIA.RefreshDisplay = True
    CATIA.DisplayFileAlerts = True

    SetBusy False

    ShowStatus VisFmt(VisTxt("msgError"), message)

End Sub

Module code

HideShowToolkit

The long block of letters and numbers at the bottom is the credit image. Do not edit it.

Option Explicit

' CATIA HIDE SHOW TOOLKIT - MAIN MODULE
' Module name: HideShowToolkit
' Run CATMain to open the window.
'=========================================================

Private Const REG_APP As String = "CatiaHideShowToolkit"
Private Const REG_SEC As String = "Settings"

' "EN" or "KO"
Public gVisLang As String


'=========================================================
' ENTRY POINT
'=========================================================

Sub CATMain()

    VisLoadLanguage
    frmHideShow.Show

End Sub

'=========================================================
' BUTTONS
' 1 = Hide, 2 = Show
'=========================================================

Public Function VisRun( _
        ByVal frm As Object, _
        ByVal actionId As Long, _
        ByRef message As String) As Boolean

    Dim d As Object
    Dim sel As Object
    Dim showState As Long
    Dim total As Long
    Dim skipped As Long
    Dim picked As Long

    VisRun = False

    Set d = CATIA.ActiveDocument
    Set sel = d.Selection

    ' CATIA show states: 0 = shown, 1 = hidden
    If actionId = 1 Then
        showState = 1
    Else
        showState = 0
    End If

    ' Elements inside parts can only be reached in design mode
    If TypeName(d) = "ProductDocument" Then
        On Error Resume Next
        d.Product.ApplyWorkMode 2
        On Error GoTo 0
    End If

    CATIA.RefreshDisplay = False

    If CtlOn(frm, "chkPlanes") Then
        picked = picked + 1
        ApplyType sel, "CATPrtSearch.Plane,all", "", showState, total, skipped
    End If

    If CtlOn(frm, "chkAxis") Then
        picked = picked + 1
        ApplyType sel, "CATPrtSearch.AxisSystem,all", "", showState, total, skipped
    End If

    If CtlOn(frm, "chkSketch") Then
        picked = picked + 1
        ApplyType sel, "CATPrtSearch.Sketch,all", "", showState, total, skipped
    End If

    If CtlOn(frm, "chkPoints") Then
        picked = picked + 1
        ApplyType sel, "CATPrtSearch.Point,all", "", showState, total, skipped
    End If

    If CtlOn(frm, "chkLines") Then
        picked = picked + 1
        ApplyType sel, "CATPrtSearch.Line,all", "", showState, total, skipped
    End If

    If CtlOn(frm, "chkSets") Then
        picked = picked + 1
        ApplyType sel, "CATPrtSearch.OpenBodyFeature,all", "", showState, total, skipped
    End If

    If CtlOn(frm, "chkCst") Then
        picked = picked + 1
        ApplyType sel, "CATAsmSearch.MfConstraint,all", "CATAsmSearch.Constraint,all", showState, total, skipped
    End If

    On Error Resume Next
    sel.Clear
    On Error GoTo 0

    CATIA.RefreshDisplay = True

    If picked = 0 Then
        message = VisTxt("msgNoType")
        Exit Function
    End If

    If showState = 1 Then
        message = VisFmt(VisTxt("msgHidden"), CStr(total))
    Else
        message = VisFmt(VisTxt("msgShown"), CStr(total))
    End If

    If skipped > 0 Then message = message & VisFmt(VisTxt("msgSkipped"), CStr(skipped))

    VisRun = True

End Function


' Searches one type and hides or shows what was found. If the
' first search text is not accepted, the second one is tried.
Private Sub ApplyType( _
        ByVal sel As Object, _
        ByVal query As String, _
        ByVal fallbackQuery As String, _
        ByVal showState As Long, _
        ByRef total As Long, _
        ByRef skipped As Long)

    Dim found As Long

    found = SetShowFor(sel, query, showState)

    If found < 0 And Len(fallbackQuery) > 0 Then found = SetShowFor(sel, fallbackQuery, showState)

    If found < 0 Then
        skipped = skipped + 1
    Else
        total = total + found
    End If

End Sub


' Returns the number of elements changed, or -1 when the
' search itself failed
Private Function SetShowFor( _
        ByVal sel As Object, _
        ByVal query As String, _
        ByVal showState As Long) As Long

    On Error GoTo Failed

    sel.Clear
    sel.Search query

    SetShowFor = sel.Count

    If sel.Count > 0 Then sel.VisProperties.SetShow showState

    sel.Clear

    Exit Function

Failed:

    SetShowFor = -1

    On Error Resume Next
    sel.Clear

End Function

' Replaces characters that are not allowed in a file name
Private Function CleanName(ByVal rawName As String) As String

    Dim i As Long
    Dim ch As String
    Dim code As Long
    Dim result As String

    For i = 1 To Len(rawName)

        ch = Mid$(rawName, i, 1)
        code = AscW(ch)

        If InStr("\/:*?""<>|", ch) > 0 Or (code >= 0 And code < 32) Then
            result = result & "_"
        Else
            result = result & ch
        End If

    Next i

    If Len(result) = 0 Then result = "Unnamed"

    CleanName = result

End Function

' Splits one CSV line into its fields
Private Function CsvSplit(ByVal lineText As String) As Collection

    Dim fields As New Collection
    Dim i As Long
    Dim ch As String
    Dim cur As String
    Dim inQuotes As Boolean

    i = 1

    Do While i <= Len(lineText)

        ch = Mid$(lineText, i, 1)

        If inQuotes Then
            If ch = """" Then
                If Mid$(lineText, i + 1, 1) = """" Then
                    cur = cur & """"
                    i = i + 1
                Else
                    inQuotes = False
                End If
            Else
                cur = cur & ch
            End If
        ElseIf ch = """" Then
            inQuotes = True
        ElseIf ch = "," Then
            fields.Add cur
            cur = ""
        Else
            cur = cur & ch
        End If

        i = i + 1

    Loop

    fields.Add cur

    Set CsvSplit = fields

End Function

' Fills docs (a Dictionary) with every loaded document under prod
Private Sub CollectDocs( _
        ByVal prod As Object, _
        ByVal docs As Object, _
        ByVal wantParts As Boolean, _
        ByVal wantProducts As Boolean)

    Dim kids As Object
    Dim child As Object
    Dim d As Object
    Dim kind As String
    Dim i As Long

    On Error Resume Next
    Set kids = prod.Products
    On Error GoTo 0

    If kids Is Nothing Then Exit Sub

    For i = 1 To kids.Count

        Set child = kids.Item(i)
        Set d = DocOf(child)

        If Not d Is Nothing Then

            kind = TypeName(d)

            If kind = "PartDocument" And wantParts Then AddDoc docs, d
            If kind = "ProductDocument" And wantProducts Then AddDoc docs, d

            CollectDocs child, docs, wantParts, wantProducts

        End If

    Next i

End Sub

' Turns \uXXXX codes into real characters
Private Function U(ByVal coded As String) As String

    Dim i As Long
    Dim result As String

    i = 1

    Do While i <= Len(coded)

        If Mid$(coded, i, 2) = "\u" And i + 5 <= Len(coded) Then
            result = result & ChrW$(CLng("&H" & Mid$(coded, i + 2, 4)))
            i = i + 6
        Else
            result = result & Mid$(coded, i, 1)
            i = i + 1
        End If

    Loop

    U = result

End Function

'=========================================================
' LOGO
' The logo picture is stored below as text.
' The form shows it with:
'     VisLoadLogo Me.Controls("imgLogo")
'=========================================================

' Puts the logo into an Image control.
' Returns False if the picture could not be loaded.
Public Function VisLoadLogo(ByVal target As Object) As Boolean

    Dim data() As Byte
    Dim filePath As String
    Dim fileNo As Integer

    VisLoadLogo = False

    On Error GoTo Failed

    data = HexToBytes(LogoHex())

    ' VBA can only load a picture from a file, so the data is
    ' written to the Windows temp folder, loaded, then deleted.
    filePath = Environ$("TEMP") & "\catia_macro_logo.gif"

    If Len(Dir$(filePath)) > 0 Then Kill filePath

    fileNo = FreeFile

    Open filePath For Binary Access Write As #fileNo
    Put #fileNo, , data
    Close #fileNo

    Set target.Picture = LoadPicture(filePath)

    VisLoadLogo = True

    On Error Resume Next
    Kill filePath

    Exit Function

Failed:

    On Error Resume Next
    Close #fileNo

End Function


' Turns pairs of hex characters back into bytes
Private Function HexToBytes(ByVal hexText As String) As Byte()

    Dim count As Long
    Dim i As Long
    Dim data() As Byte

    count = Len(hexText) \ 2

    ReDim data(0 To count - 1)

    For i = 0 To count - 1
        data(i) = CByte("&H" & Mid$(hexText, i * 2 + 1, 2))
    Next i

    HexToBytes = data

End Function

Known limitations

  • It applies to everything of the ticked type, including the three main planes of every part. It does not work on a selection only.
  • Show makes every element of that type visible, including ones that were hidden on purpose before the macro was used.
  • Hiding a geometrical set hides everything inside it.
  • The search names used for assembly constraints differ between CATIA levels. The macro tries two. If neither works, the status line says one type could not be searched.
  • The assembly is switched to design mode so the elements inside parts can be found. On a large assembly this can take a while.
  • The current selection in CATIA is cleared.
  • The window is modal, so CATIA cannot be clicked while it is open. Anything that must be selected in CATIA has to be selected before running the macro.
  • If the credit image cannot be loaded, for example when the Windows temp folder is not writable, the window opens without it.
Prepared for CATIA V5 R21 and later, VBA. Hides and shows helper geometry across a whole assembly.
