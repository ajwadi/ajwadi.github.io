CATIA Hole Table

Lists every hole feature of the active part, or of every part in an assembly, with its type, diameter, depth, thread, counterbore or countersink, position and direction. Two code blocks to paste: the form and the module.

What it does

  • One part or the whole assembly. Each part is read once.
  • 22 columns: part number, body, hole name, type, diameter, depth type and depth, thread diameter, depth and pitch, head diameter, depth and angle, the X, Y, Z position, the drilling direction, and whether the hole is active.
  • Deactivated holes are left out unless you tick the box.
  • Saved as DocumentName_Holes.csv in the folder you choose, as UTF-8 so Korean headings show correctly in Excel.
  • English / Korean button in the top right corner. The language is remembered for next time.
  • Credit image centered at the bottom, stored inside the module as text, the same as your other macros.
  • Works beside your other macros. Every shared name starts with Hol, so it does not clash with the other macros in the same VBA project.

Setup

  1. In the VBA editor, use Insert, UserForm. In the Properties window set its (Name) to frmHoleTable. Leave the form empty in the designer.
  2. Open the form's code (right-click the form, View Code), select all, delete, and paste the Form code below.
  3. Use Insert, Module. In the Properties window set its (Name) to HoleTable, then paste the Module code below.
  4. Use Debug, Compile VBAProject.
  5. Open a CATPart or CATProduct, then run CATMain from the HoleTable module.

First run: this code was written and checked for structure, but not run inside CATIA. Try it on a part with one simple hole, one threaded hole and one counterbored hole, and compare each row with the hole definition in CATIA.

        Exit Sub

    ok = HolRun(Me, actionId, message)

    CATIA.DisplayFileAlerts = True

    ShowStatus message

    Exit Sub

    message = Err.Description

    On Error Resume Next

    CATIA.RefreshDisplay = True
    CATIA.DisplayFileAlerts = True

    SetBusy False

    ShowStatus HolFmt(HolTxt("msgError"), message)

Private mOut As String
Private mCount As Long


'=========================================================
' ENTRY POINT
'=========================================================

Sub CATMain()

    HolLoadLanguage
    frmHoleTable.Show

End Sub

'=========================================================
' HOLE TABLE
'=========================================================

Public Function HolRun( _
        ByVal frm As Object, _
        ByVal actionId As Long, _
        ByRef message As String) As Boolean

    Dim docs As Object
    Dim keys As Variant
    Dim d As Object
    Dim outFolder As String
    Dim target As String
    Dim includeInactive As Boolean
    Dim i As Long

    HolRun = False

    outFolder = WithSlash(CtlText(frm, "txtOut"))

    If Not FolderExists(outFolder) Then
        message = HolTxt("msgNoOut")
        Exit Function
    End If

    includeInactive = CtlOn(frm, "chkInactive")

    Set d = CATIA.ActiveDocument
    Set docs = CreateObject("Scripting.Dictionary")

    If TypeName(d) = "PartDocument" Then

        AddDoc docs, d

    ElseIf CtlIndex(frm, "cboScope") = 1 Then

        On Error Resume Next
        d.Product.ApplyWorkMode 2
        On Error GoTo 0

        CollectDocs d.Product, docs, True, False

    Else

        message = HolTxt("msgNeedPart")
        Exit Function

    End If

    mCount = 0

    mOut = Csv(HolTxt("hNo")) & "," & Csv(HolTxt("hPN")) & "," & Csv(HolTxt("hBody")) & "," & _
           Csv(HolTxt("hHole")) & "," & Csv(HolTxt("hType")) & "," & Csv(HolTxt("hDia")) & "," & _
           Csv(HolTxt("hLimit")) & "," & Csv(HolTxt("hDepth")) & "," & Csv(HolTxt("hThreaded")) & "," & _
           Csv(HolTxt("hThrDia")) & "," & Csv(HolTxt("hThrDepth")) & "," & Csv(HolTxt("hPitch")) & "," & _
           Csv(HolTxt("hHeadDia")) & "," & Csv(HolTxt("hHeadDepth")) & "," & Csv(HolTxt("hHeadAngle")) & "," & _
           """X (mm)"",""Y (mm)"",""Z (mm)""," & Csv(HolFmt(HolTxt("hDir"), "X")) & "," & _
           Csv(HolFmt(HolTxt("hDir"), "Y")) & "," & Csv(HolFmt(HolTxt("hDir"), "Z")) & "," & _
           Csv(HolTxt("hActive")) & vbCrLf

    keys = docs.Keys

    For i = 0 To docs.Count - 1

        Set d = docs.Item(keys(i))

        Say frm, HolTxt("stItem") & " " & d.Name

        HolesOfPart d, includeInactive

    Next i

    If mCount = 0 Then
        message = HolTxt("msgNothing")
        Exit Function
    End If

    target = outFolder & CleanName(StripExt(HolDocName())) & "_Holes.csv"

    If Not WriteTextFile(target, mOut) Then
        message = HolTxt("msgWriteFail")
        Exit Function
    End If

    If CtlOn(frm, "chkOpen") Then OpenFile target

    message = HolFmt(HolTxt("msgDone"), CStr(mCount)) & " " & target

    HolRun = True

End Function


Private Sub HolesOfPart(ByVal d As Object, ByVal includeInactive As Boolean)

    Dim prt As Object
    Dim bodies As Object
    Dim b As Object
    Dim shapes As Object
    Dim s As Object
    Dim partNo As String
    Dim i As Long
    Dim j As Long

    On Error Resume Next

    partNo = d.Product.PartNumber

    Set prt = d.Part
    Set bodies = prt.Bodies

    On Error GoTo 0

    If bodies Is Nothing Then Exit Sub

    For i = 1 To bodies.Count

        Set b = bodies.Item(i)
        Set shapes = b.Shapes

        For j = 1 To shapes.Count

            Set s = shapes.Item(j)

            If TypeName(s) = "Hole" Then HoleRow prt, partNo, b.Name, s, includeInactive

        Next j

    Next i

End Sub


' Reads one hole. Values that a hole type does not have stay empty.
Private Sub HoleRow( _
        ByVal prt As Object, _
        ByVal partNo As String, _
        ByVal bodyName As String, _
        ByVal h As Object, _
        ByVal includeInactive As Boolean)

    Dim o(2) As Variant
    Dim v(2) As Variant
    Dim inactive As Boolean
    Dim holeType As Long
    Dim limitMode As Long
    Dim threadMode As Long
    Dim holeName As String
    Dim typeText As String
    Dim limitText As String
    Dim sDia As String
    Dim sDepth As String
    Dim sThrDia As String
    Dim sThrDepth As String
    Dim sPitch As String
    Dim sHeadDia As String
    Dim sHeadDepth As String
    Dim sHeadAngle As String
    Dim threadedText As String
    Dim activeText As String

    On Error Resume Next

    inactive = False
    inactive = prt.IsInactive(h)

    If inactive And Not includeInactive Then Exit Sub

    holeName = h.Name

    ' 0 simple, 1 tapered, 2 counterbored, 3 countersunk, 4 counterdrilled
    holeType = -1
    holeType = h.Type

    Select Case holeType
        Case 0: typeText = HolTxt("tSimple")
        Case 1: typeText = HolTxt("tTapered")
        Case 2: typeText = HolTxt("tCbore")
        Case 3: typeText = HolTxt("tCsink")
        Case 4: typeText = HolTxt("tCdrill")
        Case Else: typeText = HolFmt(HolTxt("tOther"), CStr(holeType))
    End Select

    sDia = Num(h.Diameter.Value, 3)

    ' 0 blind, 1 up to next, 2 up to last, 3 up to plane, 4 up to surface
    limitMode = -1
    limitMode = h.BottomLimit.LimitMode

    Select Case limitMode
        Case 0: limitText = HolTxt("dBlind")
        Case 1: limitText = HolTxt("dUpToNext")
        Case 2: limitText = HolTxt("dUpToLast")
        Case 3: limitText = HolTxt("dUpToPlane")
        Case 4: limitText = HolTxt("dUpToSurface")
        Case Else: limitText = HolFmt(HolTxt("dOther"), CStr(limitMode))
    End Select

    If limitMode = 0 Then sDepth = Num(h.BottomLimit.Dimension.Value, 3)

    ' 0 threaded, 1 smooth
    threadMode = 1
    threadMode = h.ThreadingMode

    If threadMode = 0 Then
        threadedText = HolTxt("yes")
        sThrDia = Num(h.ThreadDiameter.Value, 3)
        sThrDepth = Num(h.ThreadDepth.Value, 3)
        sPitch = Num(h.ThreadPitch.Value, 3)
    Else
        threadedText = HolTxt("no")
    End If

    If holeType >= 2 Then sHeadDia = Num(h.HeadDiameter.Value, 3)
    If holeType = 2 Or holeType = 4 Then sHeadDepth = Num(h.HeadDepth.Value, 3)
    If holeType = 3 Or holeType = 4 Then sHeadAngle = Num(h.HeadAngle.Value, 2)

    o(0) = 0
    o(1) = 0
    o(2) = 0
    v(0) = 0
    v(1) = 0
    v(2) = 0

    h.GetOrigin o
    h.GetDirection v

    If inactive Then
        activeText = HolTxt("no")
    Else
        activeText = HolTxt("yes")
    End If

    mCount = mCount + 1

    mOut = mOut & CStr(mCount) & "," & Csv(partNo) & "," & Csv(bodyName) & "," & Csv(holeName) & "," & _
           Csv(typeText) & "," & sDia & "," & Csv(limitText) & "," & sDepth & "," & Csv(threadedText) & "," & _
           sThrDia & "," & sThrDepth & "," & sPitch & "," & sHeadDia & "," & sHeadDepth & "," & sHeadAngle & "," & _
           Num(o(0), 3) & "," & Num(o(1), 3) & "," & Num(o(2), 3) & "," & _
           Num(v(0), 4) & "," & Num(v(1), 4) & "," & Num(v(2), 4) & "," & Csv(activeText) & vbCrLf

    Err.Clear

End Sub

' "" = nothing open, otherwise Product, Part, Drawing or Other
Public Function HolDocKind() As String

    Dim t As String

    On Error Resume Next

    If CATIA.Documents.Count > 0 Then t = TypeName(CATIA.ActiveDocument)

    On Error GoTo 0

    Select Case t
        Case "ProductDocument": HolDocKind = "Product"
        Case "PartDocument": HolDocKind = "Part"
        Case "DrawingDocument": HolDocKind = "Drawing"
        Case "": HolDocKind = ""
        Case Else: HolDocKind = "Other"
    End Select

End Function

' Replaces characters that are not allowed in a file name
Private Function CleanName(ByVal rawName As String) As String

    Dim i As Long
    Dim ch As String
    Dim code As Long
    Dim result As String

    For i = 1 To Len(rawName)

        ch = Mid$(rawName, i, 1)
        code = AscW(ch)

        If InStr("\/:*?""<>|", ch) > 0 Or (code >= 0 And code < 32) Then
            result = result & "_"
        Else
            result = result & ch
        End If

    Next i

    If Len(result) = 0 Then result = "Unnamed"

    CleanName = result

End Function

' A number written with a decimal point on every PC
Private Function Num(ByVal numberValue As Variant, Optional ByVal decimals As Long = 3) As String

    Dim pattern As String

    On Error Resume Next

    pattern = "0"

    If decimals > 0 Then pattern = "0." & String$(decimals, "0")

    Num = Replace(Format$(CDbl(numberValue), pattern), ",", ".")

End Function


' Saves text as UTF-8 so Excel shows Korean correctly

    Dim stm As Object

' Splits one CSV line into its fields
Private Function CsvSplit(ByVal lineText As String) As Collection

    Dim fields As New Collection
    Dim i As Long
    Dim ch As String
    Dim cur As String
    Dim inQuotes As Boolean

    i = 1

    Do While i <= Len(lineText)

        ch = Mid$(lineText, i, 1)

        If inQuotes Then
            If ch = """" Then
                If Mid$(lineText, i + 1, 1) = """" Then
                    cur = cur & """"
                    i = i + 1
                Else
                    inQuotes = False
                End If
            Else
                cur = cur & ch
            End If
        ElseIf ch = """" Then
            inQuotes = True
        ElseIf ch = "," Then
            fields.Add cur
            cur = ""
        Else
            cur = cur & ch
        End If

        i = i + 1

    Loop

    fields.Add cur

    Set CsvSplit = fields

End Function

' Fills docs (a Dictionary) with every loaded document under prod
Private Sub CollectDocs( _
        ByVal prod As Object, _
        ByVal docs As Object, _
        ByVal wantParts As Boolean, _
        ByVal wantProducts As Boolean)

    Dim kids As Object
    Dim child As Object
    Dim d As Object
    Dim kind As String
    Dim i As Long

    On Error Resume Next
    Set kids = prod.Products
    On Error GoTo 0

    If kids Is Nothing Then Exit Sub

    For i = 1 To kids.Count

        Set child = kids.Item(i)
        Set d = DocOf(child)

        If Not d Is Nothing Then

            kind = TypeName(d)

            If kind = "PartDocument" And wantParts Then AddDoc docs, d
            If kind = "ProductDocument" And wantProducts Then AddDoc docs, d

            CollectDocs child, docs, wantParts, wantProducts

        End If

    Next i

End Sub

' Turns \uXXXX codes into real characters
Private Function U(ByVal coded As String) As String

    Dim i As Long
    Dim result As String

    i = 1

    Do While i <= Len(coded)

        If Mid$(coded, i, 2) = "\u" And i + 5 <= Len(coded) Then
            result = result & ChrW$(CLng("&H" & Mid$(coded, i + 2, 4)))
            i = i + 6
        Else
            result = result & Mid$(coded, i, 1)
            i = i + 1
        End If

    Loop

    U = result

End Function

'=========================================================
' LOGO
' The logo picture is stored below as text.
' The form shows it with:
'     HolLoadLogo Me.Controls("imgLogo")
'=========================================================

' Puts the logo into an Image control.
' Returns False if the picture could not be loaded.
Public Function HolLoadLogo(ByVal target As Object) As Boolean

    Dim data() As Byte
    Dim filePath As String
    Dim fileNo As Integer

    HolLoadLogo = False

    On Error GoTo Failed

    data = HexToBytes(LogoHex())

    ' VBA can only load a picture from a file, so the data is
    ' written to the Windows temp folder, loaded, then deleted.
    filePath = Environ$("TEMP") & "\catia_macro_logo.gif"

    If Len(Dir$(filePath)) > 0 Then Kill filePath

    fileNo = FreeFile

    Open filePath For Binary Access Write As #fileNo
    Put #fileNo, , data
    Close #fileNo

    Set target.Picture = LoadPicture(filePath)

    HolLoadLogo = True

    On Error Resume Next
    Kill filePath

    Exit Function

Failed:

    On Error Resume Next
    Close #fileNo

End Function


' Turns pairs of hex characters back into bytes
Private Function HexToBytes(ByVal hexText As String) As Byte()

    Dim count As Long
    Dim i As Long
    Dim data() As Byte

    count = Len(hexText) \ 2

    ReDim data(0 To count - 1)

    For i = 0 To count - 1
        data(i) = CByte("&H" & Mid$(hexText, i * 2 + 1, 2))
    Next i

    HexToBytes = data

End Function

Known limitations

  • Only features made with the Hole command are listed. A hole made with a pocket or a groove is not a hole feature and does not appear.
  • Holes that come from a pattern are not separate features in CATIA, so only the original hole is listed, not each copy.
  • Positions and directions are in the absolute axis system of each part, not of the assembly.
  • Depth is filled in for blind holes only. For "up to next" and the other types, CATIA has no fixed depth value.
  • The thread diameter is the value CATIA stores for the thread, the nominal diameter. The standard name such as M8 is not exported.
  • The assembly is switched to design mode when every part is read.
  • The window is modal, so CATIA cannot be clicked while it is open. Anything that must be selected in CATIA has to be selected before running the macro.
  • If the credit image cannot be loaded, for example when the Windows temp folder is not writable, the window opens without it.
Prepared for CATIA V5 R21 and later, VBA. Exports a table of hole features.
