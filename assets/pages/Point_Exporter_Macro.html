CATIA Point Exporter

Writes the X, Y, Z coordinates of points and hole centers in the active part to a CSV, ready for a CMM program or an inspection plan. Two code blocks to paste: the form and the module.

What it does

  • Selected points or every point. Select points in the tree or in 3D before running, or let the macro find every point in the part.
  • Hole centers too. Tick the box and the origin of every hole feature is added.
  • Columns: No, Name, the set or body it sits in, Type, X, Y and Z in mm.
  • Decimal places from 0 to 6.
  • Saved as PartName_Points.csv in the folder you choose.
  • English / Korean button in the top right corner. The language is remembered for next time.
  • Credit image centered at the bottom, stored inside the module as text, the same as your other macros.
  • Works beside your other macros. Every shared name starts with Pt, so it does not clash with the other macros in the same VBA project.

Setup

  1. In the VBA editor, use Insert, UserForm. In the Properties window set its (Name) to frmPointExporter. Leave the form empty in the designer.
  2. Open the form's code (right-click the form, View Code), select all, delete, and paste the Form code below.
  3. Use Insert, Module. In the Properties window set its (Name) to PointExporter, then paste the Module code below.
  4. Use Debug, Compile VBAProject.
  5. Open a CATPart, select points if you want only some of them, then run CATMain from the PointExporter module.

First run: this code was written and checked for structure, but not run inside CATIA. Try it on a part with two or three points whose coordinates you know.

Private Sub SetBusy(ByVal busy As Boolean)

    Dim c As Object

    mBusy = busy

    For Each c In Me.Controls
        If TypeName(c) = "CommandButton" Then c.Enabled = Not busy
    Next c

    Me.Repaint

    DoEvents

'=========================================================
' BUTTONS
'=========================================================

Private Sub DoAction(ByVal actionId As Long)

    Dim message As String
    Dim ok As Boolean

    If mBusy Then Exit Sub

    If Not DocReady() Then
        ShowDocStatus
        Exit Sub
    End If

    SaveInputs

    SetBusy True

    ShowStatus PtTxt("stWorking")

    On Error GoTo Failed

    ok = PtRun(Me, actionId, message)

    On Error Resume Next

    CATIA.RefreshDisplay = True
    CATIA.DisplayFileAlerts = True

    SetBusy False

    ShowStatus message

    Exit Sub

Failed:

    message = Err.Description

    On Error Resume Next

    CATIA.RefreshDisplay = True
    CATIA.DisplayFileAlerts = True

    SetBusy False

    ShowStatus PtFmt(PtTxt("msgError"), message)

End Sub

Module code

PointExporter

The long block of letters and numbers at the bottom is the credit image. Do not edit it.

Option Explicit

'=========================================================
' CATIA POINT EXPORTER - MAIN MODULE
' Module name: PointExporter
' Run CATMain to open the window.
'=========================================================

Private Const REG_APP As String = "CatiaPointExporter"
Private Const REG_SEC As String = "Settings"

' "EN" or "KO"
Public gPtLang As String


'=========================================================
' ENTRY POINT
'=========================================================

Sub CATMain()

    PtLoadLanguage
    frmPointExporter.Show

End Sub

'=========================================================
' EXPORT
'=========================================================

Public Function PtRun( _
        ByVal frm As Object, _
        ByVal actionId As Long, _
        ByRef message As String) As Boolean

    Dim d As Object
    Dim prt As Object
    Dim spa As Object
    Dim sel As Object
    Dim picked As New Collection
    Dim obj As Object
    Dim meas As Object
    Dim bodies As Object
    Dim b As Object
    Dim s As Object
    Dim pt(2) As Variant
    Dim outFolder As String
    Dim target As String
    Dim content As String
    Dim itemName As String
    Dim parentName As String
    Dim decimals As Long
    Dim i As Long
    Dim j As Long
    Dim rowNo As Long
    Dim skipped As Long
    Dim ok As Boolean
    Dim searched As Boolean

    PtRun = False

    outFolder = WithSlash(CtlText(frm, "txtOut"))

    If Not FolderExists(outFolder) Then
        message = PtTxt("msgNoOut")
        Exit Function
    End If

    decimals = CLng(Val(CtlText(frm, "txtDec")))

    If decimals < 0 Then decimals = 0
    If decimals > 6 Then decimals = 6

    Set d = CATIA.ActiveDocument
    Set prt = d.Part
    Set spa = d.GetWorkbench("SPAWorkbench")
    Set sel = d.Selection

    If CtlIndex(frm, "cboSource") = 1 Then
        On Error Resume Next
        sel.Clear
        sel.Search "CATPrtSearch.Point,all"
        On Error GoTo 0
        searched = True
    End If

    ' Copy the selection first, so it can be cleared safely
    For i = 1 To sel.Count
        picked.Add sel.Item(i).Value
    Next i

    If searched Then sel.Clear

    content = Csv(PtTxt("hNo")) & "," & Csv(PtTxt("hName")) & "," & Csv(PtTxt("hParent")) & "," & _
              Csv(PtTxt("hType")) & ",""X (mm)"",""Y (mm)"",""Z (mm)""" & vbCrLf

    '---------------------------------------
    ' Points
    '---------------------------------------

    For i = 1 To picked.Count

        Set obj = picked.Item(i)

        ok = False
        itemName = ""
        parentName = ""

        On Error Resume Next

        Err.Clear

        Set meas = spa.GetMeasurable(prt.CreateReferenceFromObject(obj))

        meas.GetPoint pt

        ok = (Err.Number = 0)

        Err.Clear

        itemName = obj.Name
        parentName = obj.Parent.Parent.Name

        On Error GoTo 0

        If ok Then

            rowNo = rowNo + 1

            content = content & CStr(rowNo) & "," & Csv(itemName) & "," & Csv(parentName) & "," & _
                      Csv(PtTxt("tPoint")) & "," & Num(pt(0), decimals) & "," & _
                      Num(pt(1), decimals) & "," & Num(pt(2), decimals) & vbCrLf

        Else

            skipped = skipped + 1

        End If

    Next i

    '---------------------------------------
    ' Hole centers
    '---------------------------------------

    If CtlOn(frm, "chkHoles") Then

        Set bodies = prt.Bodies

        For i = 1 To bodies.Count

            Set b = bodies.Item(i)

            For j = 1 To b.Shapes.Count

                Set s = b.Shapes.Item(j)

                If TypeName(s) = "Hole" Then

                    ok = False

                    On Error Resume Next

                    Err.Clear

                    s.GetOrigin pt

                    ok = (Err.Number = 0)

                    Err.Clear

                    On Error GoTo 0

                    If ok Then

                        rowNo = rowNo + 1

                        content = content & CStr(rowNo) & "," & Csv(s.Name) & "," & Csv(b.Name) & "," & _
                                  Csv(PtTxt("tHole")) & "," & Num(pt(0), decimals) & "," & _
                                  Num(pt(1), decimals) & "," & Num(pt(2), decimals) & vbCrLf

                    End If

                End If

            Next j

        Next i

    End If

    If rowNo = 0 Then
        message = PtTxt("msgNothing")
        Exit Function
    End If

    target = outFolder & CleanName(StripExt(d.Name)) & "_Points.csv"

    If Not WriteTextFile(target, content) Then
        message = PtTxt("msgWriteFail")
        Exit Function
    End If

    If CtlOn(frm, "chkOpen") Then OpenFile target

    message = PtFmt(PtTxt("msgDone"), CStr(rowNo), CStr(skipped)) & " " & target

    PtRun = True

End Function

' "" = nothing open, otherwise Product, Part, Drawing or Other
Public Function PtDocKind() As String

    Dim t As String

    On Error Resume Next

    If CATIA.Documents.Count > 0 Then t = TypeName(CATIA.ActiveDocument)

    On Error GoTo 0

    Select Case t
        Case "ProductDocument": PtDocKind = "Product"
        Case "PartDocument": PtDocKind = "Part"
        Case "DrawingDocument": PtDocKind = "Drawing"
        Case "": PtDocKind = ""
        Case Else: PtDocKind = "Other"
    End Select

End Function

    On Error Resume Next

' Replaces characters that are not allowed in a file name
Private Function CleanName(ByVal rawName As String) As String

    Dim i As Long
    Dim ch As String
    Dim code As Long
    Dim result As String

    For i = 1 To Len(rawName)

        ch = Mid$(rawName, i, 1)
        code = AscW(ch)

        If InStr("\/:*?""<>|", ch) > 0 Or (code >= 0 And code < 32) Then
            result = result & "_"
        Else
            result = result & ch
        End If

    Next i

    If Len(result) = 0 Then result = "Unnamed"

    CleanName = result

End Function


Private Function StripExt(ByVal fileName As String) As String

    Dim p As Long

    If p > 1 Then
        StripExt = Left$(fileName, p - 1)

        StripExt = fileName
    End If

End Function

' One CSV field, always in quotes
Private Function Csv(ByVal fieldText As String) As String

' A number written with a decimal point on every PC
Private Function Num(ByVal numberValue As Variant, Optional ByVal decimals As Long = 3) As String

    Dim pattern As String

    On Error Resume Next

    pattern = "0"

    If decimals > 0 Then pattern = "0." & String$(decimals, "0")

    Num = Replace(Format$(CDbl(numberValue), pattern), ",", ".")

End Function


' Saves text as UTF-8 so Excel shows Korean correctly
Private Function WriteTextFile(ByVal filePath As String, ByVal content As String) As Boolean

    Dim stm As Object

    On Error GoTo Failed

    Set stm = CreateObject("ADODB.Stream")

    stm.Type = 2
    stm.Charset = "utf-8"
    stm.Open
    stm.WriteText content
    stm.SaveToFile filePath, 2
    stm.Close

    WriteTextFile = True

    Exit Function

Failed:

    On Error Resume Next
    stm.Close

End Function


Private Function ReadTextFile(ByVal filePath As String) As String

    Dim stm As Object

    On Error GoTo Failed

    Set stm = CreateObject("ADODB.Stream")

    stm.Type = 2
    stm.Charset = "utf-8"
    stm.Open
    stm.LoadFromFile filePath

    ReadTextFile = stm.ReadText

    stm.Close

    Exit Function

Failed:

    On Error Resume Next
    stm.Close

End Function


' Splits one CSV line into its fields
Private Function CsvSplit(ByVal lineText As String) As Collection

    Dim fields As New Collection
    Dim i As Long
    Dim ch As String
    Dim cur As String
    Dim inQuotes As Boolean

    i = 1

    Do While i <= Len(lineText)

        ch = Mid$(lineText, i, 1)

        If inQuotes Then
            If ch = """" Then
                If Mid$(lineText, i + 1, 1) = """" Then
                    cur = cur & """"
                    i = i + 1
                Else
                    inQuotes = False
                End If
            Else
                cur = cur & ch
            End If
        ElseIf ch = """" Then
            inQuotes = True
        ElseIf ch = "," Then
            fields.Add cur
            cur = ""
        Else
            cur = cur & ch
        End If

        i = i + 1

    Loop

    fields.Add cur

    Set CsvSplit = fields

End Function

' Fills docs (a Dictionary) with every loaded document under prod
Private Sub CollectDocs( _
        ByVal prod As Object, _
        ByVal docs As Object, _
        ByVal wantParts As Boolean, _
        ByVal wantProducts As Boolean)

    Dim kids As Object
    Dim child As Object
    Dim d As Object
    Dim kind As String
    Dim i As Long

    On Error Resume Next
    Set kids = prod.Products
    On Error GoTo 0

    If kids Is Nothing Then Exit Sub

    For i = 1 To kids.Count

        Set child = kids.Item(i)
        Set d = DocOf(child)

        If Not d Is Nothing Then

            kind = TypeName(d)

            If kind = "PartDocument" And wantParts Then AddDoc docs, d
            If kind = "ProductDocument" And wantProducts Then AddDoc docs, d

            CollectDocs child, docs, wantParts, wantProducts

        End If

    Next i

End Sub

' Turns \uXXXX codes into real characters
Private Function U(ByVal coded As String) As String

    Dim i As Long
    Dim result As String

    i = 1

    Do While i <= Len(coded)

        If Mid$(coded, i, 2) = "\u" And i + 5 <= Len(coded) Then
            result = result & ChrW$(CLng("&H" & Mid$(coded, i + 2, 4)))
            i = i + 6
        Else
            result = result & Mid$(coded, i, 1)
            i = i + 1
        End If

    Loop

    U = result

End Function

'=========================================================
' LOGO
' The logo picture is stored below as text.
' The form shows it with:
'     PtLoadLogo Me.Controls("imgLogo")
'=========================================================

' Puts the logo into an Image control.
' Returns False if the picture could not be loaded.
Public Function PtLoadLogo(ByVal target As Object) As Boolean

    Dim data() As Byte
    Dim filePath As String
    Dim fileNo As Integer

    PtLoadLogo = False

    On Error GoTo Failed

    data = HexToBytes(LogoHex())

    ' VBA can only load a picture from a file, so the data is
    ' written to the Windows temp folder, loaded, then deleted.
    filePath = Environ$("TEMP") & "\catia_macro_logo.gif"

    If Len(Dir$(filePath)) > 0 Then Kill filePath

    fileNo = FreeFile

    Open filePath For Binary Access Write As #fileNo
    Put #fileNo, , data
    Close #fileNo

    Set target.Picture = LoadPicture(filePath)

    PtLoadLogo = True

    On Error Resume Next
    Kill filePath

    Exit Function

Failed:

    On Error Resume Next
    Close #fileNo

End Function


' Turns pairs of hex characters back into bytes
Private Function HexToBytes(ByVal hexText As String) As Byte()

    Dim count As Long
    Dim i As Long
    Dim data() As Byte

    count = Len(hexText) \ 2

    ReDim data(0 To count - 1)

    For i = 0 To count - 1
        data(i) = CByte("&H" & Mid$(hexText, i * 2 + 1, 2))
    Next i

    HexToBytes = data

End Function

Known limitations

  • Coordinates are in the absolute axis system of the part. A local axis system or the assembly position is not taken into account.
  • It works on a CATPart that is the active document, not on a part inside an open assembly window.
  • Points inside sketches are not found by "Every point in the active part". Select them one by one instead, or create 3D points.
  • Holes that come from a pattern are not separate hole features in CATIA, so only the original hole is listed.
  • With "Every point in the active part", the current selection is cleared.
  • The window is modal, so CATIA cannot be clicked while it is open. Anything that must be selected in CATIA has to be selected before running the macro.
  • If the credit image cannot be loaded, for example when the Windows temp folder is not writable, the window opens without it.
